     1→// RELION task registry — comprehensive knowledge base of all RELION GUI tasks.
     2→// Reference: RELION 3.1 / 4.0 (https://github.com/3dem/relion)
     3→//
     4→// Each task definition includes:
     5→//   - key: canonical RELION task identifier
     6→//   - name, category, stage, icon
     7→//   - description: human-readable summary
     8→//   - inputs: upstream job types this task depends on
     9→//   - outputs: what this task produces (for downstream tasks / agent decisions)
    10→//   - parameters: structured list of relion-style parameters with defaults
    11→//   - decisionHints: when the agent should run this and what decisions it makes
    12→//   - typicalDuration: simulated seconds for the executor
    13→//
    14→// Categories: prep | picking | classify | refine | analyze | utility
    15→
    16→export type ParamType = "string" | "int" | "float" | "bool" | "select" | "path" | "range";
    17→
    18→export interface TaskParam {
    19→  key: string;
    20→  label: string;
    21→  type: ParamType;
    22→  default: string | number | boolean;
    23→  options?: string[];
    24→  help: string;
    25→  group?: string;
    26→  advanced?: boolean;
    27→}
    28→
    29→export interface TaskOutput {
    30→  key: string;
    31→  label: string;
    32→  kind: "movies" | "micrographs" | "particles" | "star" | "map" | "mask" | "halfmap" | "log" | "metadata";
    33→  metrics?: string[];
    34→}
    35→
    36→export interface RelionTask {
    37→  key: string;
    38→  name: string;
    39→  short: string;
    40→  category: "prep" | "picking" | "classify" | "refine" | "analyze" | "utility";
    41→  stage: number;
    42→  icon: string;
    43→  color: string;
    44→  description: string;
    45→  inputs: { types: string[]; label: string; optional?: boolean }[];
    46→  outputs: TaskOutput[];
    47→  parameters: TaskParam[];
    48→  decisionHints: {
    49→    when: string;
    50→    decides: string;
    51→  };
    52→  typicalDuration: number;
    53→  relionBinary: string;
    54→  docUrl: string;
    55→}
