   142→  interface Events {
   143→    /**
   144→     * Allow, deny, or ask before dispatch. `next()` delegates to allow; missing
   145→     * approval support turns `ask` into denial. Async gates must observe
   146→     * `exec.signal`; the registry rechecks cancellation after they settle but
   147→     * never abandons their promise.
   148→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's calls.
   149→     * @param exec - the pending call (name, parsed arguments, caller agent).
   150→     * @mode waterfall
   151→     */
   152→    'tools/pre-execute'(this: Scoped<ToolRuntime>, exec: ToolExecution, next: () => Promise<PreToolDecision>): Promise<PreToolDecision>
   153→    /**
   154→     * Around-dispatch waterfall for timeout, retry, or metrics. `next()` returns
   155→     * a normalized result; wrappers may change only `exec.signal`, while call
   156→     * identity remains immutable. The registry re-fuses the original caller
   157→     * signal before the body, so replacement cannot detach caller cancellation;
   158→     * wrappers must still restore their signal and reach quiescence.
   159→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's calls.
   160→     * @param exec - the allowed call about to dispatch (name, parsed arguments, caller agent, signal).
   161→     * @mode waterfall
   162→     */
   163→    'tools/execute'(this: Scoped<ToolRuntime>, exec: ToolDispatchExecution, next: () => Promise<ToolExecutionResult>): Promise<ToolExecutionResult>
   164→    /**
   165→     * Accept, replace, enrich, or block a normalized dispatch result. `next()`
   166→     * accepts it unchanged; thrown tools still reach this waterfall as errors. Async
   167→     * listeners must observe `exec.signal`; after they settle, caller
   168→     * cancellation replaces only a successful accepted outcome with the code
   169→     * selected by whether the tool body was invoked.
   170→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's calls.
   171→     * @param exec - the call that just ran (name, parsed arguments, caller agent).
   172→     * @param result - the dispatch outcome a listener may accept, replace, or block.
   173→     * @mode waterfall
   174→     */
   175→    'tools/post-execute'(this: Scoped<ToolRuntime>, exec: ToolExecution, result: Readonly<ToolExecutionResult>, next: () => Promise<PostToolDecision>): Promise<PostToolDecision>
   176→    /**
   177→     * Allow a listener to replace content in the DURABLE LOG COPY of one
   178→     * `run_code` sub-dispatch outcome before the bridge appends its
   179→     * `tool/code-dispatch` event. `next()` keeps the
   180→     * content unchanged; a listener may return replacement blocks (e.g. the
   181→     * spill policy's preview + locator for an oversized text result). Only the
   182→     * logged copy is affected — the program already received the complete
   183→     * value, and the model sees neither. A throwing listener is contained:
   184→     * the bridge falls back to logging the original settled content.
   185→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): agent-scoped listeners receive only that agent's dispatches.
   186→     * @param dispatch - the parent execution, sub-call identity, and the settled content to log.
   187→     * @mode waterfall
   188→     */
   189→    'tools/code-dispatch-log'(this: Scoped<ToolRuntime>, dispatch: CodeDispatchLog, next: () => Promise<ContentBlock[]>): Promise<ContentBlock[]>
   190→    /**
   191→     * Observe the frozen, lossless-JSON final outcome. Listener failures are contained.
   192→     * Scope-filtered dispatch (`@deepseek-ai/dsh-scope`): keyed by `exec.agent`.
   193→     * @param exec - the execution object that traversed the pipeline.
   194→     * @param result - a deep-frozen snapshot of the final returned result.
   195→     * @mode emit
   196→     */
   197→    'tools/result'(this: Scoped<ToolRuntime>, exec: Readonly<ToolExecution>, result: Readonly<ToolExecutionResult>): undefined
   198→    /**
   199→     * A tool was registered or unregistered, or a scoped restriction changed
   200→     * (the available tool set changed — possibly for one scope only). An
   201→     * UNFILTERED registry-subject notification, deliberately not scope-filtered
   202→     * dispatch: a global change concerns every agent's next assembly, so a
   203→     * scoped listener subscribing here sees every change, not just its own
   204→     * scope's.
   205→     * @mode emit
   206→     */
   207→    'tools/change'(): void
   208→  }

   221→/** A registered tool: its schema plus the execution function. */
   222→export interface ToolDefinition extends ToolSchema {
   223→  /** Mandatory canonical output declaration. */
   224→  readonly output: ToolOutputDefinition
   225→  /**
   226→   * Run one accepted call and return only its canonical lossless-JSON value.
   227→   * Async work must observe or forward `exec.signal` and settle only after its
   228→   * owned work reaches quiescence. The registry preserves caller cancellation
   229→   * through around-dispatch signal replacement and does not abandon this
   230→   * promise, but it cannot hard-kill same-process code.
   231→   * @param args - losslessly snapshotted, frozen model arguments.
   232→   * @param exec - execution identity, cancellation signal, and context deferral.
   233→   * @returns the canonical value declared by `output.schema`.
   234→   */
   235→  execute(args: unknown, exec: ToolRunContext): Promise<unknown>
   236→  /**
   237→   * Synchronous last-mile transform for model-facing content. The registry
   238→   * snapshots this callback when execution starts and invokes it exactly once
   239→   * for every normalized outcome, including pipeline failures that bypass
   240→   * `tools/post-execute`, immediately before lossless materialization.
   241→   * Returning `undefined` preserves the content; every other result field
   242→   * remains registry-owned. The callback must be total and must not throw.
   243→   * @param exec - immutable execution identity and arguments.
   244→   * @param result - complete normalized outcome before materialization.
   245→   * @returns replacement content, or `undefined` to preserve it.
   246→   */
   247→  finalizeContent?(exec: Readonly<ToolExecution>, result: Readonly<ToolExecutionResult>): ContentBlock[] | undefined
   248→  /**
   249→   * Cooperative tool-call timeout budget in milliseconds. Omit for no deadline.
   250→   * Enforced by `@deepseek-ai/dsh-tool-call-timeout-policy` (a `tools/execute` wrapper); it
   251→   * is NEVER sent to the model — `schemas()` whitelists only name/description/
   252→   * parameters. Declaring it asserts this tool forwards `exec.signal` to a
   253→   * cooperative implementation that can reach quiescence when the signal aborts.
   254→   */
   255→  timeoutMs?: number
   256→  /**
   257→   * Pure synchronous classifier for overlap with sibling tool calls. Only
   258→   * `true` opts in; omission, exceptions, non-`true` returns, and invalid
   259→   * `defineTool` arguments are exclusive. This metadata is never model-visible.
   260→   *
   261→   * Opted-in executions must not mutate parent-owned state. Shared state must
   262→   * tolerate concurrent dispatch; recorder races are permitted only when they
   263→   * commute or fail closed. See the
   264→   * [parallel-tool-call Agent Note](../../../../.agents/notes/implemented/feature/2026-07-10-parallel-tool-call-execution.md)
   265→   * for the full contract.
   266→   * @param args - parsed arguments; `defineTool` validates before calling.
   267→   * @returns Whether this call may join a parallel group.
   268→   */
   269→  isConcurrencySafe?(args: unknown): boolean
   270→  /**
   271→   * Optional: how to present the PENDING state of one call in a UI, derived from
   272→   * the call's `args` (parsed arguments, `unknown` — the tool validates/narrows
   273→   * its own input). Returns a {@link ToolCallView} (a `card`-tagged render intent),
   274→   * or `undefined` (or omit the method) to fall back to a generic presentation
   275→   * (title = tool name, raw args as input). Pure and side-effect-free: a UI may
   276→   * call it during live streaming AND a session-log replay, so it must depend
   277→   * only on `args`.
   278→   */
   279→  presentCall?(args: unknown): ToolCallView | undefined
   280→  /**
   281→   * Optional: how to present the COMPLETED state, given the same `args` and the
   282→   * durable result projection (`content`, failure state, and optional `meta`). Returns a
   283→   * {@link ToolResultView}, or `undefined` (or omit the method) to keep the
   284→   * pending title and render the raw result content. Pure and side-effect-free
   285→   * for the same replay reason.
   286→   */
   287→  presentResult?(args: unknown, result: ToolResult): ToolResultView | undefined
   288→}

   488→/**
   489→ * Thrown (internally) when the model requests a tool that isn't registered.
   490→ * Extends {@link HarnessError} (`code: 'UNKNOWN_TOOL'`) so an unknown-tool
   491→ * failure is as routable as a tool-thrown one — retry/sandbox/replay code can
   492→ * distinguish it from a tool body's own error.
   493→ */
   494→export class ToolNotFoundError extends HarnessError {
   495→  /**
   496→   * @param toolName - the name the caller asked for.
   497→   * @param reachableFrom - how the model reaches this tool instead, when the
   498→   *   name IS visible and only the presentation denies calling it directly.
   499→   *   Omitted for a name that is registered nowhere.
   500→   */
   501→  constructor(toolName: string, reachableFrom?: string) {
   502→    super(
   503→      reachableFrom === undefined
   504→        ? `unknown tool "${toolName}"`
   505→        : `unknown tool "${toolName}": ${reachableFrom}`,
   506→      'UNKNOWN_TOOL',
   507→    )
   508→    this.name = 'ToolNotFoundError'
   509→  }

   524→/** Convert one projector exception into the canonical invalid-output failure. */

   526→  return new ToolOutputError(toolName, [`output.${projector} failed: ${errorMessage(error)}`])

   529→/** Snapshot one projector result before later durable-result materialization. */
   530→function snapshotProjection<T>(toolName: string, projector: 'render' | 'presentationMeta', candidate: T): T {
   531→  try {
   532→    const detached = snapshotJsonValue(candidate)
   533→    if (detached === undefined) {
   534→      throw new ToolOutputError(toolName, [`output.${projector} returned non-lossless JSON`])
   535→    }
   536→    return detached
   537→  } catch (error: unknown) {
   538→    if (error instanceof ToolOutputError) throw error
   539→    throw projectionError(toolName, projector, error)
   540→  }
   541→}

   543→/** Snapshot one body or policy value into the canonical invalid-output failure class. */
   544→function snapshotToolValue(toolName: string, candidate: unknown): JsonValue {
   545→  try {
   546→    const detached = snapshotJsonValue(candidate)
   547→    if (detached === undefined) throw new ToolOutputError(toolName, ['value is not lossless JSON'])
   548→    return detached as JsonValue
   549→  } catch (error: unknown) {
   550→    if (error instanceof ToolOutputError) throw error
   551→    throw new ToolOutputError(toolName, [`value snapshot failed: ${errorMessage(error)}`])
   552→  }

   582→/**
   583→ * Pre-dispatch decision. `allow` runs the call; `deny` materializes an error;
   584→ * `ask` runs only after an approval service returns `allowed-once` and otherwise
   585→ * denies. Input rewriting is excluded because arguments are already logged and
   586→ * presented.

   590→  | { kind: 'deny'; reason: string }
   591→  | { kind: 'ask'; reason?: string }

   593→/**
   594→ * Post-dispatch decision: accept, replace one projection, attach context for the
   595→ * next request, or block by turning corrective feedback into an error result.
   596→ */
   597→export type PostToolDecision =
   598→  | { kind: 'accept'; content?: ContentBlock[]; value?: never; additionalContexts?: UserMessage[] }
   599→  | { kind: 'accept'; value: JsonValue; content?: never; additionalContexts?: UserMessage[] }
   600→  | { kind: 'block'; feedback: ContentBlock[]; additionalContexts?: UserMessage[] }
   601→
   602→/**
   603→ * Best-effort human-readable message from an arbitrary thrown value: Error
   604→ * instances use `.message`; non-Error objects with a string `message`
   605→ * property (e.g. `throw { message: 'denied' }`) use it too; everything else
   606→ * is stringified.
   607→ */
   608→function errorMessage(error: unknown): string {
   609→  try {
   610→    if (error instanceof Error) return error.message
   611→    if (typeof error === 'object' && error !== null
   612→      && 'message' in error && typeof error.message === 'string') {
   613→      return error.message
   614→    }
   615→    return String(error)
   616→  } catch {
   617→    // A hostile thrown value can trap `instanceof`, property access, or string
   618→    // coercion. Error normalization is the outermost safety boundary, so its
   619→    // fallback must itself be total.
   620→    return '<unprintable thrown value>'
   621→  }
   622→}
   623→
   624→/** Derive one failure message from policy feedback without changing its rendered blocks. */
   625→function failureMessageFromContent(content: ContentBlock[]): string {
   626→  const text = content
   627→    .map(block => block.type === 'text' ? block.text : `[${block.type} content]`)
   628→    .join('\n')
   629→  return text.length > 0 ? text : 'tool result blocked by post-execute policy'
   630→}

   632→/** Snapshot and freeze one durable tool-result projection or reject lossy data. */

   634→  const detached = snapshotJsonValue(candidate)
   635→  if (detached === undefined) {
   636→    throw new TypeError('tool result must be losslessly JSON-serializable')
   637→  }
   638→  return deepFreeze(detached)
   639→}

   713→/** One scope's complete tool-registry contribution. */
   714→class ToolLayer implements ScopeLayer {
   715→  readonly tools: NamedEntries<ToolDefinition>
   716→  readonly restrictions = new AnonymousEntries<CompiledToolRestriction>()
   717→  readonly guards = new AnonymousEntries<ToolGuard>()
   718→  /**
   719→   * Presentation this scope's agent declared for itself, shadowing the
   720→   * deployment default. One cell rather than an entry table: two answers to
   721→   * "which form does the model see" is a contradiction, not a merge.
   722→   */
   723→  mode: ToolPresentationMode | undefined
   724→
   725→  constructor(scope: ScopeKey | undefined) {
   726→    this.tools = new NamedEntries(name => new Error(scope === undefined
   727→      ? `tool "${name}" is already registered (for a per-agent variant, register through that agent's \`agent.ctx\` instead)`
   728→      : `tool "${name}" is already registered in this scope`))
   729→  }
   730→
   731→  /** Whether every contribution table in this aggregate layer is empty. */
   732→  isEmpty(): boolean {
   733→    return this.tools.isEmpty() && this.restrictions.isEmpty() && this.guards.isEmpty()
   734→      && this.mode === undefined
   735→  }
   736→
   737→  /** Whether every compiled restriction in this layer admits a global tool name. */
   738→  admits(name: string): boolean {
   739→    for (const filter of this.restrictions.values()) {
   740→      if ((filter.allow !== undefined && !filter.allow.has(name))
   741→        || (filter.deny !== undefined && filter.deny.has(name))) return false
   742→    }
   743→    return true
   744→  }
   745→
   746→  /** First monotonic denial from this layer's live guard registrations. */
   747→  guardReason(exec: ToolExecution): string | undefined {
   748→    for (const guard of this.guards.values()) {
   749→      const reason = guard(exec)
   750→      if (reason !== undefined) return reason
   751→    }
   752→    return undefined
   753→  }
   754→}

   783→/**
   784→ * Tool registry and execution pipeline. Scoped registrations shadow globals;
   785→ * one visibility resolver feeds presentation, lookup, and dispatch.
   786→ */
   787→export class ToolRuntime extends Service {
   788→  static inject = ['systemPrompt']
   789→
   790→  static Config: z<Config> = z.object({
   791→    mode: z.union(['native', 'code', 'both'] as const).default('native'),
   792→    maxParallelSubCalls: z.natural().min(1).default(10),
   793→  })
   794→
   795→  /** Internal staged view consumed by `dsh-agent-loop`'s parallel scheduler. */
   796→  readonly [TOOL_RUNTIME_SCHEDULER]: ToolRuntimeScheduler = {
   797→    prepare: exec => this.prepareScheduledExecution(exec),
   798→    dispatch: exec => this.dispatchScheduledExecution(exec),
   799→    finalize: (exec, result) => this.finalizeScheduledExecution(exec, result),
   800→    finish: (exec, result) => this.finishScheduledExecution(exec, result),
   801→  }
   802→
   803→  /** Context deferred by a running tool body, keyed by its scheduler-owned execution. */
   804→  private deferredContexts = new WeakMap<ToolRunContext, UserMessage[]>()
   805→  /** Executions whose tool body declared the current turn complete. */
   806→  private concludingExecutions = new WeakSet<ToolExecution>()
   807→  /** Original caller cancellation, kept outside the wrapper-mutable execution object. */
   808→  private cancellationStates = new WeakMap<ToolRunContext, ToolCancellationState>()
   809→  /** Definition-owned final content transform snapshotted before policy begins. */
   810→  private contentFinalizers = new WeakMap<ToolRunContext, ToolDefinition['finalizeContent']>()
   811→  private readonly layers = new ScopedLayers(
   812→    scope => new ToolLayer(scope),
   813→    () => { this.ctx.emit('tools/change') },
   814→  )
   815→  /** Presentation for scopes that declare none; {@link presentAs} shadows it per scope. */
   816→  private readonly defaultMode: ToolPresentationMode
   817→  private readonly maxParallelSubCalls: number
   818→  /**
   819→   * Reserved presentation transport, kept outside the filterable registration
   820→   * layers. Built on first need rather than at construction: which agents run
   821→   * a code mode is no longer known when the service is constructed, and the
   822→   * transport is stateless beyond its closures over `this`.
   823→   */
   824→  private codeTransport: ToolDefinition | undefined
   825→
   826→  constructor(ctx: Context, config: Config = {}) {
   827→    super(ctx, 'tools')
   828→    // The schema already defaulted an omitted mode; the ?? narrows the
   829→    // optional-input type for direct (non-Loader) construction in tests.
   830→    this.defaultMode = config.mode ?? 'native'
   831→    this.maxParallelSubCalls = resolveMaxParallelSubCalls(config.maxParallelSubCalls)
   832→    ctx.systemPrompt.tools(context => this.wireSchemas(context.scope))
   833→    if (this.defaultMode !== 'native') {
   834→      ctx.systemPrompt.section(this.collapseSection())
   835→      ctx.systemPrompt.section(this.sdkSection())
   836→    }
   837→  }
   838→
   839→  /**
   840→   * The prompt statement of the `code` executor collapse, registered wherever
   841→   * {@link sdkSection} is and rendering empty outside an effective `code`.
   842→   *
   843→   * Every tool contributes its own guidance section naming its tool, none of
   844→   * them qualify how that tool is reached, and they all render before the SDK
   845→   * (orders 100-199 against {@link SDK_SECTION_ORDER}). Without this the model
   846→   * reads a catalog of tools it is told to use and no statement that only
   847→   * `run_code` may be called, so it emits a native call, receives
   848→   * `UNKNOWN_TOOL` for a tool the prompt just declared, and concludes the
   849→   * deployment is inconsistent. {@link COLLAPSE_SECTION_ORDER} places the rule
   850→   * before that guidance rather than after it.
   851→   *
   852→   * `both` renders empty: native calls do execute there, so the rule is false.
   853→   * @returns the section registration.
   854→   */
   855→  private collapseSection(): { name: string; order: number; text: (context: { scope?: ScopeKey }) => string } {
   856→    return {
   857→      name: 'tools:code-only',
   858→      order: COLLAPSE_SECTION_ORDER,
   859→      // The SAME predicate the executor denies by, so the prompt cannot state
   860→      // a rule the registry does not enforce (see `collapses`).
   861→      text: context => this.modeFor(context.scope) === 'code' ? CODE_ONLY_INSTRUCTION : '',
   862→    }
   863→  }
   864→
   865→  /**
   866→   * The generated-SDK prompt section, registered globally by a code-mode
   867→   * deployment and per scope by {@link presentAs}.
   868→   *
   869→   * The body regenerates from the CALLING scope, and renders empty for an
   870→   * agent presenting natively — an agent that opted out under a code-mode
   871→   * deployment still sees the global registration, and an empty section is
   872→   * dropped from the rendered prompt.
   873→   * @returns the section registration.
   874→   */
   875→  private sdkSection(): { name: string; order: number; text: (context: { scope?: ScopeKey }) => string } {
   876→    return {
   877→      name: 'tools:sdk',
   878→      order: SDK_SECTION_ORDER,
   879→      // Regenerate from the calling scope's visible tools in stable order.
   880→      text: (context) => {
   881→        const mode = this.modeFor(context.scope)
   882→        if (mode === 'native') return ''
   883→        const runtime = this.requireCodeRuntime(mode)
   884→        // Own-property read: a language like `toString`/`constructor` would
   885→        // otherwise resolve an inherited Object.prototype member as a renderer.
   886→        const render = SDK_RENDERERS[runtime.language]
   887→        /* v8 ignore next -- requireCodeRuntime rejects an unknown language before this runs. */
   888→        if (render === undefined) throw new Error(`dsh-tools: no SDK renderer for ${runtime.language}`)
   889→        return render(this.sdkSchemas(context.scope))
   890→      },
   891→    }
   892→  }
   893→
   894→  /**
   895→   * The presentation one scope's agent sees: its own declaration, else the
   896→   * deployment default.
   897→   * @param scope - the calling agent, or undefined for the global view.
   898→   * @returns the resolved presentation mode.
   899→   */
   900→  private modeFor(scope?: ScopeKey): ToolPresentationMode {
   901→    // Nearest scope wins along the chain: a preset's standing declaration
   902→    // covers every agent parented under it, and an agent's own (were one ever
   903→    // declared) would override its preset's. The mode decides what the model
   904→    // SEES, which is exactly the class of fact the chain inherits.
   905→    const layers = this.layers.chainLayers(scope)
   906→    for (let index = layers.length - 1; index >= 0; index -= 1) {
   907→      const mode = layers[index]?.mode
   908→      if (mode !== undefined) return mode
   909→    }
   910→    return this.defaultMode
   911→  }
   912→
   913→  /**
   914→   * The reserved `run_code` transport, built on first need.
   915→   *
   916→   * It never enters the global layer: per-agent restrictions must not remove
   917→   * it, and a scoped registration must not shadow it. The visibility resolver
   918→   * appends it after resolving the filterable global/scoped capability layers,
   919→   * and only for scopes whose mode actually presents it.
   920→   * @returns the shared transport definition.
   921→   */
   922→  private requireCodeTransport(): ToolDefinition {
   923→    this.codeTransport ??= createRunCodeTool(this, {
   924→      requireRuntime: () => this.requireCodeRuntime(this.defaultMode),
   925→      // The language-aware description/parameters getters read the runtime
   926→      // without demanding one, so a native-default process can still project
   927→      // the transport for an agent that chose code.
   928→      peekRuntime: () => this.ctx.get('codeRuntime'),
   929→      maxParallel: this.maxParallelSubCalls,
   930→      shapeDispatchLog: dispatch => this.shapeDispatchLog(dispatch),
   931→    })
   932→    return this.codeTransport
   933→  }
   934→
   935→  /**
   936→   * Present the calling scope's tools in `mode` instead of the deployment
   937→   * default. Nearest scope on the chain wins, so a preset's standing
   938→   * declaration covers every agent joined under it.
   939→   *
   940→   * Scoped only, and one declaration per scope: this is how an agent preset
   941→   * composes Code Mode agents beside native ones in the same process, and a
   942→   * process-global override would be the `mode` config field instead.
   943→   * @param mode - the presentation the covered agents' models see.
   944→   * @returns the exact disposer that restores the deployment default.
   945→   */
   946→  presentAs(mode: ToolPresentationMode): () => void {
   947→    const ctx = this.ctx
   948→    if (scopeOf(ctx) === undefined) {
   949→      throw new Error('tools.presentAs() requires a scoped context (agent.ctx): a context-global presentation is the `mode` config field on the tools row')
   950→    }
   951→    const dispose = ctx.effect(function* (this: ToolRuntime) {
   952→      yield this.layers.effect(
   953→        ctx,
   954→        (layer) => {
   955→          if (layer.mode !== undefined) {
   956→            throw new Error(`tools.presentAs("${mode}") conflicts with "${layer.mode}" already declared for this scope; one composition selects one presentation`)
   957→          }
   958→          layer.mode = mode
   959→          return () => { layer.mode = undefined }
   960→        },
   961→        { label: 'tools.presentAs()' },
   962→      )
   963→      // The SDK and collapse sections are per scope for the same reason the
   964→      // mode is. Under a deployment that already defaults to a code mode this
   965→      // shadows the global registration with an identical body, which costs
   966→      // nothing and keeps one rule instead of a case analysis.
   967→      if (mode !== 'native') {
   968→        yield ctx.systemPrompt.section(this.collapseSection())
   969→        yield ctx.systemPrompt.section(this.sdkSection())
   970→      }
   971→    }.bind(this), 'tools.presentAs()')
   972→    // oxlint-disable-next-line typescript/no-misused-promises -- synchronous composite teardown; direct return preserves disposer identity
   973→    return dispose
   974→  }
   975→
   976→  /**
   977→   * Build one scope's wire schemas and names for prompt-order validation.
   978→   * Restrictions do not make known tools invalid, but a mode collapse does.
   979→   */
   980→  private wireSchemas(scope?: ScopeKey): ToolProviderResult {
   981→    const view = this.view(scope)
   982→    const mode = this.modeFor(scope)
   983→    if (mode === 'native') {
   984→      const schemas = [...view.visible.values()].map(definition => this.schemaOf(definition, false))
   985→      return { schemas, knownNames: [...view.knownNames] }
   986→    }
   987→    // Validate the runtime language BEFORE projecting schemas: schemaOf reads
   988→    // run_code's language-aware description/parameters getters, whose own
   989→    // flavor-table guard would otherwise surface first. This keeps the
   990→    // renderer-table rejection the canonical assembly-time error for a
   991→    // language with no SDK renderer.
   992→    this.requireCodeRuntime(mode)
   993→    const schemas = [...view.visible.values()].map(definition => this.schemaOf(definition, false))
   994→    if (mode === 'code') {
   995→      return {
   996→        schemas: schemas.filter(schema => schema.name === RUN_CODE_NAME),
   997→        knownNames: [RUN_CODE_NAME],
   998→      }
   999→    }
  1000→    return { schemas, knownNames: [...view.knownNames, RUN_CODE_NAME] }
  1001→  }
  1002→
  1003→  /**
  1004→   * Resolve the code runtime or throw the actionable misconfiguration error.
  1005→   * Read at use time (assembly / run_code execution), NOT via static
  1006→   * `inject`: an inject entry would hold `ctx.tools` — and every tool plugin
  1007→   * behind it — hostage to a code runtime existing even under `mode:
  1008→   * 'native'` (the loop's optional-backend idiom, same as
  1009→   * `sessionPersistence`).
  1010→   *
  1011→   * Assembly and `run_code` execution read separately, so the language is not
  1012→   * bound to a request. Harmless while one published backend exists — both
  1013→   * reads return the same flavor — but a reload that swapped in a second
  1014→   * language between them would hand a program written against one SDK to the
  1015→   * other. Binding it is deferred until a second backend ships (the first
  1016→   * point it is testable); rationale in the
  1017→   * [language-dispatch note](../../../../.agents/notes/implemented/feature/2026-07-31-code-mode-language-dispatch.md).
  1018→   */
  1019→  private requireCodeRuntime(mode: ToolPresentationMode): CodeRuntime {
  1020→    const runtime = this.ctx.get('codeRuntime')
  1021→    if (!runtime) {
  1022→      throw new Error(`dsh-tools: mode "${mode}" requires a code runtime — load a ctx.codeRuntime implementation (e.g. @deepseek-ai/dsh-code-runtime-worker-thread) or set tools mode to "native"`)
  1023→    }
  1024→    if (!Object.hasOwn(SDK_RENDERERS, runtime.language)) {
  1025→      const known = Object.keys(SDK_RENDERERS).map(name => JSON.stringify(name)).join(', ')
  1026→      throw new Error(`dsh-tools: no SDK renderer registered for runtime language ${JSON.stringify(runtime.language)} (known: ${known})`)
  1027→    }
  1028→    return runtime
  1029→  }

  1031→  /**
  1032→   * Register globally or in the calling agent scope. Scoped tools shadow
  1033→   * globals; duplicates within one layer and the reserved `run_code` name fail.
  1034→   * @param definition - tool schema, execution, and optional finalization/presentation callbacks.
  1035→   * @returns the exact disposer that unregisters the tool.
  1036→   */
  1037→  register(definition: ToolDefinition): () => void {
  1038→    const name = definition.name
  1039→    const output = (definition as Partial<ToolDefinition>).output
  1040→    if (output === undefined || typeof output !== 'object'
  1041→      || typeof output.render !== 'function'
  1042→      || (output.presentationMeta !== undefined && typeof output.presentationMeta !== 'function')) {
  1043→      throw new TypeError(`tool "${name}" must declare output { schema, render, presentationMeta? }`)
  1044→    }
  1045→    assertSupportedJsonSchema(output.schema)
  1046→    const timeoutMs = definition.timeoutMs
  1047→    if (timeoutMs !== undefined
  1048→      && (!Number.isFinite(timeoutMs) || timeoutMs <= 0)) {
  1049→      throw new TypeError(`tool "${name}" timeoutMs must be a positive finite number`)
  1050→    }
  1051→    // Reserved unconditionally: any agent may select a code mode for itself,
  1052→    // so a name free to take under the deployment default would become a
  1053→    // collision the moment a preset mounted.
  1054→    if (name === RUN_CODE_NAME) {
  1055→      throw new Error(`tool name "${RUN_CODE_NAME}" is reserved for the Code Mode presentation transport and cannot be registered or shadowed`)
  1056→    }
  1057→    return this.layers.effect(
  1058→      this.ctx,
  1059→      layer => layer.tools.insert(name, definition),
  1060→      { label: 'tools.register()' },
  1061→    )
  1062→  }
  1063→
  1064→  /**
  1065→   * Restrict global tools for the calling agent scope. Empty filters, unknown
  1066→   * names, scope-local names, and reserved transport names fail. Restrictions
  1067→   * intersect; scoped registrations remain visible.
  1068→   * @param filter - global-tool mask: `allow` (keep only) and/or `deny` (remove).
  1069→   * @returns the exact disposer that lifts this restriction.
  1070→   */
  1071→  restrict(filter: ToolRestriction): () => void {
  1072→    const scope = scopeOf(this.ctx)
  1073→    if (scope === undefined) {
  1074→      throw new Error('tools.restrict() requires a scoped context (agent.ctx): a context-global restriction would mask every agent — deny the tool for the intended agent instead')
  1075→    }
  1076→    const allow = filter.allow
  1077→    const deny = filter.deny
  1078→    if (allow === undefined && deny === undefined) {
  1079→      throw new Error('tools.restrict({}) is a no-op: pass `allow` and/or `deny` (an empty filter is almost always a materialized-empty-config bug)')
  1080→    }
  1081→    const compiled: CompiledToolRestriction = {
  1082→      ...allow !== undefined ? { allow: new Set(allow) } : {},
  1083→      ...deny !== undefined ? { deny: new Set(deny) } : {},
  1084→    }
  1085→    if ([...allow ?? [], ...deny ?? []].includes(RUN_CODE_NAME)) {
  1086→      throw new Error(`tools.restrict() cannot name reserved Code Mode presentation transport "${RUN_CODE_NAME}"; restrict end-capability tools instead`)
  1087→    }
  1088→    const known = this.view(scope).restrictableNames
  1089→    const unknown = [...allow ?? [], ...deny ?? []].filter(name => !known.has(name))
  1090→    if (unknown.length > 0) {
  1091→      throw new Error(`tools.restrict() names unknown global tool${unknown.length > 1 ? 's' : ''} ${unknown.map(n => `"${n}"`).join(', ')}; known global tools: ${[...known].sort().join(', ') || '(none)'}`)
  1092→    }
  1093→    return this.layers.effect(
  1094→      this.ctx,
  1095→      layer => layer.restrictions.append(compiled),
  1096→      { label: 'tools.restrict()' },
  1097→    )
  1098→  }
  1099→
  1100→  /**
  1101→   * Register a monotonic guard after the extensible `tools/pre-execute`
  1102→   * waterfall. A plain-context guard applies globally; one registered through
  1103→   * `agent.ctx` applies only to that agent. Any matching guard may deny by
  1104→   * returning a reason, while no guard can force-allow a call another guard
  1105→   * denied. The exact effect disposer is returned for ordered ownership and
  1106→   * HMR cleanup.
  1107→   * @param guard - synchronous check; a returned string denies the execution.
  1108→   * @returns the exact disposer that unregisters the guard.
  1109→   */
  1110→  guard(guard: ToolGuard): () => void {
  1111→    return this.layers.effect(
  1112→      this.ctx,
  1113→      layer => layer.guards.append(guard),
  1114→      { label: 'tools.guard()', notify: false },
  1115→    )
  1116→  }

  1118→  /** First monotonic denial from the global then the scope chain's guard layers, farthest first. */
  1119→  private guardReason(exec: ToolExecution): string | undefined {
  1120→    const globalReason = this.layers.global.guardReason(exec)
  1121→    if (globalReason !== undefined) return globalReason
  1122→    if (exec.agent === undefined) return undefined
  1123→    for (const layer of this.layers.chainLayers(exec.agent)) {
  1124→      const reason = layer.guardReason(exec)
  1125→      if (reason !== undefined) return reason
  1126→    }
  1127→    return undefined
  1128→  }
  1129→
  1130→  /**
  1131→   * Resolve every registry fact one scope needs in one layer traversal. The
  1132→   * visible map applies restrictions to the INHERITED surface, then the
  1133→   * scope's own registrations and the reserved presentation transport; the
  1134→   * other sets retain the pre-restriction facts needed by restriction and
  1135→   * prompt-order validation.
  1136→   *
  1137→   * A restriction filters what a scope inherits — the global layer and every
  1138→   * ancestor layer on its chain — and never what its OWN layer registers.
  1139→   * That exemption is what a per-child capability filter has to keep intact:
  1140→   * the delegation runtime registers a child's reporting and structured-output
  1141→   * tools into the child's own layer, and a filter naming the capabilities the
  1142→   * child may use must not strip the machinery it answers through.
  1143→   *
  1144→   * Reading the exempt set as "the global layer" instead of "not mine" held
  1145→   * only while every model-facing tool sat in the host composition. Once
  1146→   * presets moved them onto the agent plane they became an ANCESTOR
  1147→   * contribution, so a child's filter silently stopped constraining anything
  1148→   * it was given.
  1149→   * @param scope - the viewing scope (the agent), or undefined for the global view.
  1150→   * @returns the complete derived view for that scope.
  1151→   */
  1152→  private view(scope?: ScopeKey): ToolView {
  1153→    // Scope-chain layers, farthest ancestor first, the exact scope last.
  1154→    const layers = this.layers.chainLayers(scope)
  1155→    // Chain-blind on purpose: this is the ONE layer whose registrations the
  1156→    // scope owns rather than inherits, and it is absent until the scope
  1157→    // contributes something.
  1158→    const own = this.layers.peek(scope)
  1159→    // Inherited surface, nearest ancestor last: a nearer scope's same-name
  1160→    // entry shadows a farther one, and the global layer is the farthest.
  1161→    const inherited = new Map<string, ToolDefinition>(this.layers.global.tools.entries())
  1162→    for (const layer of layers) {
  1163→      if (layer === own) continue
  1164→      for (const [name, definition] of layer.tools.entries()) inherited.set(name, definition)
  1165→    }
  1166→    const visible = new Map<string, ToolDefinition>()
  1167→    const knownNames = new Set<string>()
  1168→    const restrictableNames = new Set<string>()
  1169→    for (const [name, definition] of inherited) {
  1170→      knownNames.add(name)
  1171→      restrictableNames.add(name)
  1172→      // Restrictions intersect across the whole chain: any scope on it may
  1173→      // mask an inherited name for everything nested inside it.
  1174→      if (layers.every(layer => layer.admits(name))) visible.set(name, definition)
  1175→    }
  1176→    // The scope's own registrations last, shadowing an inherited name and
  1177→    // outside the filter above.
  1178→    if (own !== undefined) {
  1179→      for (const [name, definition] of own.tools.entries()) {
  1180→        knownNames.add(name)
  1181→        visible.set(name, definition)
  1182→      }
  1183→    }
  1184→    // Presentation infrastructure is resolved last and outside capability
  1185→    // filtering. Registration rejects this reserved name, so the insertion is
  1186→    // an invariant assertion as well as protection against future layer
  1187→    // changes. Per scope: a native agent must not find `run_code` in its
  1188→    // dispatch table because some other agent in the process presents it.
  1189→    if (this.modeFor(scope) !== 'native') {
  1190→      visible.set(RUN_CODE_NAME, this.requireCodeTransport())
  1191→    }
  1192→    return { visible, knownNames, restrictableNames }
  1193→  }

  1195→  /**
  1196→   * Look up a tool as one scope sees it (scoped
  1197→   * shadows global; a restricted-away global reads as absent). Presenters pass
  1198→   * the calling agent so the rendered card matches the definition that
  1199→   * actually executed.
  1200→   * @param name - the tool name as registered.
  1201→   * @param scope - the viewing scope (the agent); omitted = the global view.
  1202→   * @returns the definition the scope resolves, or undefined when none is visible.
  1203→   */
  1204→  get(name: string, scope?: ScopeKey): ToolDefinition | undefined {
  1205→    return this.view(scope).visible.get(name)
  1206→  }

  1208→  /**
  1209→   * Resolve the definition that MAY EXECUTE for a call, applying the mode
  1210→   * collapse at the operation boundary that owns it. The registry view
  1211→   * (`get`) is presentation-agnostic; here a MODEL-DIRECT call under `code`
  1212→   * may only name the reserved `run_code` transport, while a nested
  1213→   * sub-dispatch (a `parent` token set — the `run_code` SDK calling a tool
  1214→   * it bound) may call any visible tool. Denial surfaces as `UNKNOWN_TOOL`
  1215→   * through the executor, matching an absent definition.
  1216→   * @param name - the tool name as registered.
  1217→   * @param scope - the viewing scope (the agent); omitted = the global view.
  1218→   * @param nested - whether the call is a transport sub-dispatch, not a model-direct call.
  1219→   * @returns the definition that may run, or undefined when the call must be rejected.
  1220→   */
  1221→  private resolveExecution(name: string, scope: ScopeKey | undefined, nested: boolean): ToolDefinition | undefined {
  1222→    const tool = this.get(name, scope)
  1223→    if (tool === undefined) return undefined
  1224→    if (this.collapses(name, scope, nested)) return undefined
  1225→    return tool
  1226→  }

  1234→  schemas(scope?: ScopeKey): ToolSchema[] {

  1238→  /** Project visible callable tools onto the generated Code Mode SDK contract. */

  1241→      .filter(definition => definition.name !== RUN_CODE_NAME)

  1255→  /** Project one definition onto the model-facing schema fields. */
  1256→  private schemaOf(definition: ToolDefinition, detachParameters: boolean): ToolSchema {
  1257→    const { name, description, parameters } = definition
  1258→    const detached = detachParameters ? snapshotJsonValue(parameters) : parameters
  1259→    if (detached === undefined) {
  1260→      throw new Error(`tool "${name}" parameters must be lossless JSON before schema projection`)

  1263→      name,
  1264→      description,
  1265→      parameters: detached,

  1269→  /**
  1270→   * Classify a pending call through the caller's visible tool definition. Only
  1271→   * an exact `true` is parallel; unknown, hidden, undeclared, invalid, or
  1272→   * throwing classifiers are exclusive.
  1273→   * @param exec - call name, parsed arguments, and optional agent scope.
  1274→   * @returns the fail-closed scheduling mode.
  1275→   */
  1276→  executionMode(exec: ToolExecutionInput): ToolExecutionMode {
  1277→    const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1278→    if (!tool?.isConcurrencySafe) return { kind: 'exclusive' }
  1279→    try {
  1280→      const concurrencySafe: unknown = tool.isConcurrencySafe(exec.arguments)
  1281→      return concurrencySafe === true ? { kind: 'parallel' } : { kind: 'exclusive' }
  1282→    } catch {
  1283→      return { kind: 'exclusive' }
  1284→    }
  1285→  }

  1287→  /**
  1288→   * Run the `tools/code-dispatch-log` waterfall over one settled sub-dispatch
  1289→   * and return the content the bridge should log on `tool/code-dispatch`.
  1290→   * Contained: when a listener throws, the method logs the original settled
  1291→   * content; that failure must not fail the dispatch or omit the settle event. Private:
  1292→   * the ONE consumer is the `run_code` bridge this registry constructs, which
  1293→   * receives it as a capability parameter (the `requireRuntime` idiom) — the
  1294→   * waterfall, not this invoker, is the public extension point.
  1295→   */
  1296→  private async shapeDispatchLog(dispatch: CodeDispatchLog): Promise<ContentBlock[]> {
  1297→    try {
  1298→      return await this.ctx.waterfall(
  1299→        scopeTarget(this, dispatch.agent), 'tools/code-dispatch-log', dispatch,
  1300→        () => Promise.resolve(dispatch.content),
  1301→      )
  1302→    } catch (error: unknown) {
  1303→      this.ctx.logger.warn(`tools: code-dispatch-log listener failed for ${dispatch.name}: ${errorMessage(error)}; logging the original settled content`)
  1304→      return dispatch.content
  1305→    }
  1306→  }

  1308→  /**
  1309→   * Whether the `code` mode collapse denies a model-direct call: only the
  1310→   * reserved `run_code` transport may be named. Nested sub-dispatches (a
  1311→   * `parent` token set) bypass the collapse. One home for the
  1312→   * security-relevant predicate, shared by {@link resolveExecution} and
  1313→   * {@link createExecution} so the two can never drift apart.
  1314→   *
  1315→   * Resolved through {@link modeFor}, NOT `defaultMode`: an agent given `code`
  1316→   * by an agent preset under a native deployment is the composition
  1317→   * `dsh-agent-tool-presentation` exists for, and reading the deployment default would
  1318→   * leave exactly that agent uncollapsed — announcing one surface while
  1319→   * executing another, which is the bypass this collapse closes.
  1320→   * @param name - the tool name as registered.
  1321→   * @param scope - the viewing scope whose effective presentation mode applies.
  1322→   * @param nested - whether the call is a transport sub-dispatch, not a model-direct call.
  1323→   */
  1324→  private collapses(name: string, scope: ScopeKey | undefined, nested: boolean): boolean {
  1325→    return !nested && this.modeFor(scope) === 'code' && name !== RUN_CODE_NAME
  1326→  }
  1327→
  1328→  /**
  1329→   * Execute through pre-policy, guards, around-dispatch, post-policy,
  1330→   * definition-owned content finalization, and final notification. Tool and
  1331→   * listener failures resolve as materialized error results; an invisible tool
  1332→   * reports `UNKNOWN_TOOL`. The returned outcome is the same lossless, frozen
  1333→   * snapshot final observers receive. Cancellation
  1334→   * arriving after entry and before final result materialization skips a
  1335→   * not-yet-started body with `ABORTED_BEFORE_DISPATCH` or replaces a
  1336→   * successful started outcome with `ABORTED`; already-started work is still
  1337→   * drained and may retain a tool-owned structured error.
  1338→   * @param exec - the typed same-process call input. The registry assigns its
  1339→   *   correlation token before policy begins.
  1340→   * @returns the materialized final result.
  1341→   */
  1342→  async execute(exec: ToolExecutionInput): Promise<ToolExecutionResult> {
  1343→    return this.prepareExecution(exec, prepared => this.completeScheduledExecution(prepared))
  1344→  }

  1346→  private async completeScheduledExecution(prepared: ScheduledToolPreparation): Promise<ToolExecutionResult> {
  1347→    switch (prepared.kind) {
  1348→      case 'dispatch': {
  1349→        const dispatched = await this.dispatchScheduledExecution(prepared.exec)
  1350→        return dispatched.kind === 'post-result'
  1351→          ? await this.finalizeScheduledExecution(prepared.exec, dispatched.result)
  1352→          : this.finishScheduledExecution(prepared.exec, dispatched.result)
  1353→      }
  1354→      case 'post-result':
  1355→        return await this.finalizeScheduledExecution(prepared.exec, prepared.result)
  1356→      case 'final-result':
  1357→        return this.finishScheduledExecution(prepared.exec, prepared.result)
  1358→      /* v8 ignore next -- closed-union exhaustiveness guard */
  1359→      default:
  1360→        return assertNever(prepared, 'scheduled tool preparation')
  1361→    }
  1362→  }

  1364→  private createExecution(exec: ToolExecutionInput): ScheduledToolPreparation | { kind: 'ready'; exec: MutableToolRunContext } {
  1365→    const deferredContexts: UserMessage[] = []
  1366→    const token = createExecutionToken()
  1367→    const callId = exec.callId
  1368→    const rootCallId = exec.rootCallId ?? callId
  1369→    const name = exec.name
  1370→    const agent = exec.agent
  1371→    const parent = exec.parent
  1372→    const signal = exec.signal
  1373→    // Distinguish a mode-collapsed call (visible in the scope, denied only by
  1374→    // the `code` collapse) from a genuinely unknown tool. A collapsed call is
  1375→    // deterministically denied, so it terminates BEFORE the extensible policy
  1376→    // pipeline: pre-execute listeners, approval `ask`, and guards must never
  1377→    // observe — or worse, approve — a call that can only fail. An unknown tool
  1378→    // keeps the historical dispatch-stage `UNKNOWN_TOOL` path so policy
  1379→    // listeners still see every name that reaches the registry.
  1380→    const visible = this.get(name, agent)
  1381→    const collapsed = visible !== undefined && this.collapses(name, agent, parent !== undefined)
  1382→    const concludingExecutions = this.concludingExecutions
  1383→    const base = {
  1384→      token,
  1385→      callId,
  1386→      rootCallId,
  1387→      name,
  1388→      signal,
  1389→      ...agent !== undefined ? { agent } : {},
  1390→      ...parent !== undefined ? { parent } : {},
  1391→      deferContext(context: UserMessage): void {
  1392→        deferredContexts.push(context)
  1393→      },
  1394→      concludeTurn(): void {
  1395→        concludingExecutions.add(this as unknown as ToolExecution)
  1396→      },
  1397→    }
  1398→    // Capture the finalizer BEFORE argument materialization: the
  1399→    // `finalizeContent` contract snapshots the callback when the call starts,
  1400→    // and an arguments getter can replace or clear the registered callback
  1401→    // during `snapshotJsonValue`. The collapse only decides whether the
  1402→    // CAPTURED callback is retained: the pre-dispatch abort path keeps it
  1403→    // (the cancellation contract routes aborted results through it — a getter
  1404→    // that aborts mid-materialization before an invalid-args failure lands in
  1405→    // the same retained path), while the `UNKNOWN_TOOL` denial and the
  1406→    // invalid-args failure of a NON-ABORTED collapsed call drop it (the call
  1407→    // could never execute).
  1408→    const capturedFinalizer = visible?.finalizeContent?.bind(visible)
  1409→    const finalizerFor = (): ToolDefinition['finalizeContent'] | undefined =>
  1410→      collapsed && !signal.aborted ? undefined : capturedFinalizer
  1411→    try {
  1412→      const detached = snapshotJsonValue(exec.arguments)
  1413→      if (detached === undefined) {
  1414→        throw new TypeError('tool execution arguments must be losslessly JSON-serializable')
  1415→      }
  1416→      const execution: MutableToolRunContext = { ...base, arguments: deepFreeze(detached) }
  1417→      this.deferredContexts.set(execution, deferredContexts)
  1418→      this.contentFinalizers.set(execution, finalizerFor())
  1419→      this.cancellationStates.set(execution, {
  1420→        callerSignal: signal,
  1421→        bodyInvoked: false,
  1422→      })
  1423→      if (collapsed) {
  1424→        // The collapse denies the call before the policy pipeline, but a
  1425→        // pre-dispatch abort still keeps the established cancellation
  1426→        // contract: `prepare`'s caller-cancellation check is skipped for
  1427→        // final-results, so honor the abort here instead of surfacing
  1428→        // `UNKNOWN_TOOL` on an already-cancelled call.
  1429→        if (signal.aborted) {
  1430→          return { kind: 'final-result', exec: execution, result: toolAbortedBeforeDispatchResult() }
  1431→        }
  1432→        // The name IS visible here, so the denial carries the route the model
  1433→        // must take instead. Without it the model reads a bare `unknown tool`
  1434→        // for a tool the prompt just declared and concludes the deployment is
  1435→        // broken rather than correcting itself.
  1436→        return {
  1437→          kind: 'final-result',
  1438→          exec: execution,
  1439→          result: toolErrorResult(new ToolNotFoundError(
  1440→            name,
  1441→            `only \`${RUN_CODE_NAME}\` is callable directly — call \`${name}\` from inside a \`${RUN_CODE_NAME}\` program instead`,
  1442→          )),
  1443→        }
  1444→      }
  1445→      return { kind: 'ready', exec: execution }
  1446→    } catch (error: unknown) {
  1447→      const execution: MutableToolRunContext = { ...base, arguments: undefined }
  1448→      this.contentFinalizers.set(execution, finalizerFor())
  1449→      return { kind: 'final-result', exec: execution, result: toolErrorResult(error) }
  1450→    }
  1451→  }

  1453→  /**
  1454→   * Run the ordered pre-execute and monotonic guard stages for the scheduler.
  1455→   * @param input - the caller-supplied execution input.
  1456→   * @returns the prepared execution plus the next scheduler stage.
  1457→   * @internal
  1458→   */
  1459→  private async prepareScheduledExecution(input: ToolExecutionInput): Promise<ScheduledToolPreparation> {
  1460→    return this.prepareExecution(input, prepared => prepared)
  1461→  }
  1462→
  1463→  private async prepareExecution<T>(
  1464→    input: ToolExecutionInput,
  1465→    next: (prepared: ScheduledToolPreparation) => T | PromiseLike<T>,
  1466→  ): Promise<T> {
  1467→    const created = this.createExecution(input)
  1468→    if (created.kind !== 'ready') return next(created)
  1469→    const exec = created.exec
  1470→    if (this.callerCancelled(exec)) {
  1471→      return next({ kind: 'final-result', exec, result: toolAbortedBeforeDispatchResult() })
  1472→    }
  1473→    try {
  1474→      const carrier = scopeTarget(this, exec.agent)
  1475→      const gate = await this.ctx.waterfall(
  1476→        carrier, 'tools/pre-execute', exec,
  1477→        () => Promise.resolve<PreToolDecision>({ kind: 'allow' }),
  1478→      )
  1479→      const askResolution: ToolAskResolution = gate.kind === 'ask'
  1480→        ? await this.serviceAsk(exec, gate)
  1481→        : { decision: gate, approvalCancelled: false }
  1482→      const { decision } = askResolution
  1483→      if (this.callerCancelled(exec) && askResolution.approvalCancelled) {
  1484→        return await next({ kind: 'post-result', exec, result: toolAbortedBeforeDispatchResult() })
  1485→      }
  1486→      const denialReason = decision.kind === 'allow'
  1487→        ? this.guardReason(exec)
  1488→        : decision.reason
  1489→      if (denialReason !== undefined) {
  1490→        return await next({
  1491→          kind: 'post-result',
  1492→          exec,
  1493→          result: this.materializeFinalResult({
  1494→            content: [{ type: 'text', text: `Error: ${denialReason}` }],
  1495→            isError: true,
  1496→            error: { message: denialReason },
  1497→          }),
  1498→        })
  1499→      }
  1500→      if (this.callerCancelled(exec)) {
  1501→        return await next({ kind: 'post-result', exec, result: toolAbortedBeforeDispatchResult() })
  1502→      }
  1503→      return await next({ kind: 'dispatch', exec })
  1504→    } catch (error: unknown) {
  1505→      return next({ kind: 'final-result', exec, result: toolErrorResult(error) })
  1506→    }
  1507→  }
  1508→
  1509→  /** Whether the original caller signal is currently aborted. */
  1510→  private callerCancelled(exec: ToolRunContext): boolean {
  1511→    const state = this.cancellationStates.get(exec)
  1512→    /* v8 ignore next -- only registry-minted executions reach the staged scheduler methods */
  1513→    if (state === undefined) throw new Error('tool registry scheduler invariant violated: missing cancellation state')
  1514→    return state.callerSignal.aborted
  1515→  }
  1516→
  1517→  /** Canonical cancellation outcome selected by whether the tool body started. */
  1518→  private cancellationResult(exec: ToolRunContext, prior?: ToolExecutionResult): ToolExecutionResult {
  1519→    const state = this.cancellationStates.get(exec)
  1520→    /* v8 ignore next -- only registry-minted executions reach the staged scheduler methods */
  1521→    if (state === undefined) throw new Error('tool registry scheduler invariant violated: missing cancellation state')
  1522→    return state.bodyInvoked
  1523→      ? toolAbortedResult(prior)
  1524→      : toolAbortedBeforeDispatchResult(prior)
  1525→  }
  1526→
  1527→  /**
  1528→   * Dispatch the registered body with the original caller signal fused back
  1529→   * into any around-wrapper replacement. Cancellation never abandons the body:
  1530→   * a started promise reaches quiescence before its outcome becomes `ABORTED`.
  1531→   */
  1532→  private async dispatchToolBody(exec: MutableToolRunContext): Promise<ToolExecutionResult> {
  1533→    const state = this.cancellationStates.get(exec)
  1534→    /* v8 ignore next -- only registry-minted executions reach the staged scheduler methods */
  1535→    if (state === undefined) throw new Error('tool registry scheduler invariant violated: missing cancellation state')
  1536→    const wrapperSignal = exec.signal
  1537→    const fused = fuseToolSignals(state.callerSignal, wrapperSignal)
  1538→    const signal = fused.signal
  1539→
  1540→    if (isAborted(signal)) {
  1541→      fused.dispose()
  1542→      return toolAbortedBeforeDispatchResult()
  1543→    }
  1544→    exec.signal = signal
  1545→    try {
  1546→      const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1547→      if (!tool) throw new ToolNotFoundError(exec.name)
  1548→      state.bodyInvoked = true
  1549→      const returned = await tool.execute(exec.arguments, exec)
  1550→      const result = this.createSuccessResult(exec, tool, returned)
  1551→      return isAborted(signal)
  1552→        ? toolAbortedResult(result)
  1553→        : result
  1554→    } catch (error: unknown) {
  1555→      return toolErrorResult(error)
  1556→    } finally {
  1557→      fused.dispose()
  1558→      exec.signal = wrapperSignal
  1559→    }
  1560→  }
  1561→
  1562→  /**
  1563→   * Run around-dispatch and the tool body. Tool and unknown-tool failures still
  1564→   * receive post-execute; pipeline failures are already final.
  1565→   * @param exec - the prepared execution.
  1566→   * @returns whether the result still needs post-execute.
  1567→   * @internal
  1568→   */
  1569→  private async dispatchScheduledExecution(exec: ToolRunContext): Promise<ScheduledToolDispatch> {
  1570→    try {
  1571→      const mutableExec = exec as MutableToolRunContext
  1572→      const carrier = scopeTarget(this, exec.agent)
  1573→      const result = await this.ctx.waterfall(
  1574→        carrier, 'tools/execute', mutableExec,
  1575→        () => this.dispatchToolBody(mutableExec),
  1576→      )
  1577→      const normalized = this.normalizeDispatchResult(exec, result)
  1578→      const deferredContexts = this.deferredContexts.get(exec)
  1579→      /* v8 ignore next -- dispatch only receives executions minted by this registry's prepare stage */
  1580→      if (deferredContexts === undefined) throw new Error('tool registry scheduler invariant violated: unprepared execution')
  1581→      const resultWithDeferredContexts: ToolExecutionResult = deferredContexts.length === 0
  1582→        ? normalized
  1583→        : this.markCanonical(exec, {
  1584→          ...normalized,
  1585→          additionalContexts: [
  1586→            ...deferredContexts,
  1587→            ...normalized.additionalContexts ?? [],
  1588→          ],
  1589→        })
  1590→      return {
  1591→        kind: 'post-result',
  1592→        result: this.callerCancelled(exec) && !resultWithDeferredContexts.isError
  1593→          ? this.cancellationResult(exec, resultWithDeferredContexts)
  1594→          : resultWithDeferredContexts,
  1595→      }
  1596→    } catch (error: unknown) {
  1597→      return { kind: 'final-result', result: toolErrorResult(error) }
  1598→    }
  1599→  }
  1600→
  1601→  /**
  1602→   * Run ordered post-execute, then apply definition-owned content finalization,
  1603→   * materialize, and notify the final outcome.
  1604→   * @param exec - the prepared execution.
  1605→   * @param result - dispatch/pre result that still needs post-execute.
  1606→   * @returns the materialized final result.
  1607→   * @internal
  1608→   */
  1609→  private async finalizeScheduledExecution(exec: ToolRunContext, result: ToolExecutionResult): Promise<ToolExecutionResult> {
  1610→    try {
  1611→      const postResult = await this.postExecute(exec, result)
  1612→      return this.finishScheduledExecution(
  1613→        exec,
  1614→        this.callerCancelled(exec) && !postResult.isError
  1615→          ? this.cancellationResult(exec, postResult)
  1616→          : postResult,
  1617→      )
  1618→    } catch (error: unknown) {
  1619→      return this.finishScheduledExecution(exec, toolErrorResult(error))
  1620→    }
  1621→  }
  1622→
  1623→  /**
  1624→   * Materialize the candidate, apply definition-owned content finalization,
  1625→   * then materialize and notify the authoritative result.
  1626→   * @param exec - the prepared execution.
  1627→   * @param result - final result.
  1628→   * @returns the materialized final result.
  1629→   * @internal
  1630→   */
  1631→  private finishScheduledExecution(exec: ToolRunContext, result: ToolExecutionResult): ToolExecutionResult {
  1632→    let materializedResult: ToolExecutionResult
  1633→    try {
  1634→      materializedResult = this.materializeFinalResult(result)
  1635→    } catch (error: unknown) {
  1636→      materializedResult = this.materializeFinalResult(toolErrorResult(error))
  1637→    }
  1638→    let finalResult: ToolExecutionResult
  1639→    try {
  1640→      finalResult = this.materializeFinalResult(this.applyFinalContent(exec, materializedResult))
  1641→    } catch (error: unknown) {
  1642→      finalResult = this.materializeFinalResult(toolErrorResult(error))
  1643→    }
  1644→    this.notifyResult(exec, finalResult)
  1645→    return finalResult
  1646→  }
  1647→
  1648→  /** Apply the snapshotted tool-owned content transform without exposing other result fields. */
  1649→  private applyFinalContent(exec: ToolRunContext, result: ToolExecutionResult): ToolExecutionResult {
  1650→    const finalizeContent = this.contentFinalizers.get(exec)
  1651→    if (finalizeContent === undefined) return result
  1652→    const content = finalizeContent(exec, result)
  1653→    return content === undefined ? result : { ...result, content }
  1654→  }
  1655→
  1656→  /** Notify observers without exposing a mutation or error channel into the outcome. */
  1657→  private notifyResult(exec: ToolExecution, result: ToolExecutionResult): void {
  1658→    // Freeze the registry's live object before observers receive its readonly
  1659→    // WeakMap-keyable view.
  1660→    Object.freeze(exec)
  1661→    const { name: toolName, callId } = exec
  1662→    const reportFailure = (error: unknown): void => {
  1663→      this.ctx.logger.warn(`tool "${toolName}" (${callId}): tools/result observer failed: ${errorMessage(error)}`)
  1664→    }
  1665→    const callbacks = this.ctx.events.dispatch('emit', [
  1666→      scopeTarget(this, exec.agent), 'tools/result', exec, result,
  1667→    ])
  1668→    for (const callback of callbacks) {
  1669→      try {
  1670→        const returned: unknown = callback(exec, result)
  1671→        void Promise.resolve(returned).catch(reportFailure)
  1672→      } catch (error: unknown) {
  1673→        reportFailure(error)
  1674→      }
  1675→    }
  1676→  }
  1677→
  1678→  /**
  1679→   * Resolve an `ask` decision to allow/deny through the approval seam. The
  1680→   * seam is consumed opportunistically with `ctx.get('approval')` — a
  1681→   * deployment that composes no ApprovalService keeps the historical degrade
  1682→   * to deny, and an unmount mid-session degrades the same way on the next ask.
  1683→   * An agent-less execution also degrades: without an agent there is no
  1684→   * session to audit to and no UI to route to. Otherwise the outcome maps
  1685→   * one-to-one — `allowed-once` proceeds; the three non-grants deny with
  1686→   * distinct reasons so the model can tell a human "no" from an absent
  1687→   * approval channel.
  1688→   */
  1689→  private async serviceAsk(
  1690→    exec: ToolExecution,
  1691→    ask: Extract<PreToolDecision, { kind: 'ask' }>,
  1692→  ): Promise<ToolAskResolution> {
  1693→    const approval = this.ctx.get('approval')
  1694→    if (approval === undefined) {
  1695→      return {
  1696→        decision: { kind: 'deny', reason: ask.reason ?? `tool "${exec.name}" requires approval (not yet supported)` },
  1697→        approvalCancelled: false,
  1698→      }
  1699→    }
  1700→    if (exec.agent === undefined) {
  1701→      return {
  1702→        decision: { kind: 'deny', reason: `tool "${exec.name}" requires approval, but the call has no agent to route it through` },
  1703→        approvalCancelled: false,
  1704→      }
  1705→    }
  1706→    const outcome = await approval.request({
  1707→      agent: exec.agent,
  1708→      toolName: exec.name,
  1709→      callId: exec.callId,
  1710→      ...ask.reason !== undefined ? { reason: ask.reason } : {},
  1711→      signal: exec.signal,
  1712→    })
  1713→    switch (outcome) {
  1714→      case 'allowed-once': return { decision: { kind: 'allow' }, approvalCancelled: false }
  1715→      case 'rejected': return {
  1716→        decision: { kind: 'deny', reason: `the user rejected tool "${exec.name}"` },
  1717→        approvalCancelled: false,
  1718→      }
  1719→      case 'cancelled': return {
  1720→        decision: { kind: 'deny', reason: `approval for tool "${exec.name}" was cancelled` },
  1721→        approvalCancelled: true,
  1722→      }
  1723→      case 'unavailable': return {
  1724→        decision: { kind: 'deny', reason: `tool "${exec.name}" requires approval, but no approval channel is available` },
  1725→        approvalCancelled: false,
  1726→      }
  1727→      default: return assertNever(outcome, 'ApprovalOutcome')
  1728→    }
  1729→  }
  1730→
  1731→  /**
  1732→   * Run the `tools/post-execute` waterfall over a dispatched `result` and apply
  1733→   * its {@link PostToolDecision}: `accept` keeps the call successful (replacing
  1734→   * `content` when given), `block` turns it into an `isError` whose content is
  1735→   * the corrective `feedback`. Either decision may attach `additionalContexts`,
  1736→   * which are ferried on the returned result for the loop's active-batch FIFO.
  1737→   * Context deferred by the tool body survives an accepted result but is
  1738→   * discarded when the outer call is blocked; a block exposes only context the
  1739→   * blocking decision explicitly supplied.
  1740→   * Runs inside `execute`'s outer try/catch (a throwing listener → isError).
  1741→   */
  1742→  private async postExecute(exec: ToolExecution, result: ToolExecutionResult): Promise<ToolExecutionResult> {
  1743→    const decision = await this.ctx.waterfall(
  1744→      scopeTarget(this, exec.agent), 'tools/post-execute', exec, result,
  1745→      () => Promise.resolve<PostToolDecision>({ kind: 'accept' }),
  1746→    )
  1747→    const decisionContexts = decision.additionalContexts ?? []
  1748→    if (decision.kind === 'block') {
  1749→      const message = failureMessageFromContent(decision.feedback)
  1750→      return this.markCanonical(exec, {
  1751→        content: decision.feedback,
  1752→        isError: true,
  1753→        error: { message },
  1754→        ...decisionContexts.length > 0 ? { additionalContexts: decisionContexts } : {},
  1755→      })
  1756→    }
  1757→    if (Object.hasOwn(decision, 'content') && Object.hasOwn(decision, 'value')) {
  1758→      throw new TypeError('tools/post-execute accept decision cannot replace both value and content')
  1759→    }
  1760→    const additionalContexts = [
  1761→      ...result.additionalContexts ?? [],
  1762→      ...decisionContexts,
  1763→    ]
  1764→    if (Object.hasOwn(decision, 'value')) {
  1765→      if (result.isError) {
  1766→        throw new TypeError('tools/post-execute cannot replace the value of a failed result')
  1767→      }
  1768→      const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1769→      if (tool === undefined) throw new ToolNotFoundError(exec.name)
  1770→      const replaced = this.createSuccessResult(exec, tool, decision.value)
  1771→      return this.markCanonical(exec, {
  1772→        ...replaced,
  1773→        ...additionalContexts.length > 0 ? { additionalContexts } : {},
  1774→      })
  1775→    }
  1776→    return this.markCanonical(exec, {
  1777→      ...result,
  1778→      ...decision.content !== undefined ? { content: decision.content } : {},
  1779→      ...additionalContexts.length > 0 ? { additionalContexts } : {},
  1780→    })
  1781→  }
  1782→
  1783→  /** Registry-normalized results and the exact dispatch that validated each value. */
  1784→  private readonly canonicalResults = new WeakMap<object, ToolExecutionToken>()
  1785→
  1786→  /** Mark one registry-normalized result as canonical only for its owning dispatch. */
  1787→  private markCanonical<T extends ToolExecutionResult>(exec: ToolExecution, result: T): T {
  1788→    this.canonicalResults.set(result, exec.token)
  1789→    return result
  1790→  }

  1792→  /** Snapshot, validate, render, and optionally project one successful body value. */
  1793→  private createSuccessResult(exec: ToolExecution, tool: ToolDefinition, candidate: unknown): ToolExecutionSuccess {
  1794→    const detached = snapshotToolValue(tool.name, candidate)
  1795→    const violations = validateJsonSchemaValue(tool.output.schema, detached, 'value')
  1796→    if (violations.length > 0) throw new ToolOutputError(tool.name, violations)
  1797→    const value = deepFreeze(detached)
  1798→    let rendered: ContentBlock[]
  1799→    try {
  1800→      rendered = tool.output.render(exec.arguments, value)
  1801→    } catch (error: unknown) {
  1802→      throw projectionError(tool.name, 'render', error)
  1803→    }
  1804→    const content = snapshotProjection(tool.name, 'render', rendered)
  1805→    let meta: JsonValue | undefined
  1806→    if (exec.parent === undefined && tool.output.presentationMeta !== undefined) {
  1807→      let projected: JsonValue
  1808→      try {
  1809→        projected = tool.output.presentationMeta(exec.arguments, value)
  1810→      } catch (error: unknown) {
  1811→        throw projectionError(tool.name, 'presentationMeta', error)
  1812→      }
  1813→      meta = snapshotProjection(tool.name, 'presentationMeta', projected)
  1814→    }
  1815→    const concludesTurn = this.concludingExecutions.has(exec)
  1816→    return this.markCanonical(exec, this.materializeFinalResult({
  1817→      isError: false,
  1818→      value,
  1819→      content,
  1820→      ...meta !== undefined ? { meta } : {},
  1821→      ...concludesTurn ? { concludesTurn: true as const } : {},
  1822→    }) as ToolExecutionSuccess)
  1823→  }

  1825→  /** Normalize an around-dispatch wrapper's authored result through the owning output contract. */
  1826→  private normalizeDispatchResult(exec: ToolExecution, result: ToolExecutionResult): ToolExecutionResult {
  1827→    if (this.canonicalResults.get(result) === exec.token) return result
  1828→    if (result.isError) {
  1829→      return this.markCanonical(exec, {
  1830→        isError: true,
  1831→        error: result.error,
  1832→        content: result.content,
  1833→        ...result.meta !== undefined ? { meta: result.meta } : {},
  1834→        ...result.additionalContexts !== undefined ? { additionalContexts: result.additionalContexts } : {},
  1835→      })
  1836→    }
  1837→    const tool = this.resolveExecution(exec.name, exec.agent, exec.parent !== undefined)
  1838→    if (tool === undefined) throw new ToolNotFoundError(exec.name)
  1839→    const normalized = this.createSuccessResult(exec, tool, result.value)
  1840→    return this.markCanonical(exec, {
  1841→      ...normalized,
  1842→      ...result.additionalContexts !== undefined ? { additionalContexts: result.additionalContexts } : {},
  1843→    })
  1844→  }
  1845→
  1846→  /** Materialize the authoritative commit outcome once, immediately before `tools/result`. */
  1847→  private materializeFinalResult(result: ToolExecutionResult): ToolExecutionResult {
  1848→    const presentation = {
  1849→      content: result.content,
  1850→      ...result.meta !== undefined ? { meta: result.meta } : {},
  1851→      ...result.additionalContexts !== undefined ? { additionalContexts: result.additionalContexts } : {},
  1852→    }
  1853→    if (result.isError) {
  1854→      return materializePresentation({ isError: true as const, error: result.error, ...presentation })
  1855→    }
  1856→    const detached = materializePresentation({
  1857→      isError: false as const,
  1858→      ...presentation,
  1859→      ...result.concludesTurn === true ? { concludesTurn: true as const } : {},
  1860→    })
  1861→    return deepFreeze({ ...detached, value: result.value })
  1862→  }
  1863→}

  1880→/** Read live abort state across an await without treating it as synchronously immutable. */
  1881→function isAborted(signal: AbortSignal): boolean {
  1882→  return signal.aborted

  1885→/**
  1886→ * Fuse caller and wrapper cancellation without nesting `AbortSignal.any`.
  1887→ * Keeping the relay dispatch-scoped also removes listeners when work settles.
  1888→ */
  1889→function fuseToolSignals(caller: AbortSignal, wrapper: AbortSignal): FusedToolSignal {
  1890→  if (caller === wrapper) return { signal: caller, dispose() {} }
  1891→
  1892→  const controller = new AbortController()
  1893→  let listening = false
  1894→  const dispose = (): void => {
  1895→    if (!listening) return
  1896→    listening = false
  1897→    caller.removeEventListener('abort', abortFromCaller)
  1898→    wrapper.removeEventListener('abort', abortFromWrapper)
  1899→  }
  1900→  const abortFrom = (source: AbortSignal): void => {
  1901→    const reason: unknown = source.reason
  1902→    controller.abort(reason)
  1903→    dispose()
  1904→  }
  1905→  const abortFromCaller = (): void => { abortFrom(caller) }
  1906→  const abortFromWrapper = (): void => { abortFrom(wrapper) }
  1907→
  1908→  if (wrapper.aborted) abortFromWrapper()
  1909→  else if (caller.aborted) abortFromCaller()
  1910→  else {
  1911→    listening = true
  1912→    caller.addEventListener('abort', abortFromCaller, { once: true })
  1913→    wrapper.addEventListener('abort', abortFromWrapper, { once: true })
  1914→  }
  1915→  return { signal: controller.signal, dispose }
  1916→}

  1918→/** Canonical result when cancellation supersedes success after body invocation. */
  1919→function toolAbortedResult(prior?: ToolExecutionResult): ToolExecutionResult {

  1921→  return {
  1922→    content: [{ type: 'text', text: 'Error: tool call aborted' }],
  1923→    isError: true,
  1924→    error: {
  1925→      message: 'tool call aborted',
  1926→      info: { name: 'AbortError', code: TOOL_ABORTED },
