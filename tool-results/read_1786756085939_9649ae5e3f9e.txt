    92→# ---------------------------------------------------------------------------
    93→# Run a subprocess, stream stdout+stderr, capture lines
    94→# ---------------------------------------------------------------------------
    95→def run_cmd(cmd, cwd, env, on_line):
    96→    """Run a command, calling on_line(level, line) for each stdout/stderr line."""
    97→    on_line("info", "$ " + " ".join(cmd))
    98→    proc = subprocess.Popen(
    99→        cmd, cwd=cwd, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
   100→        text=True, bufsize=1,
   101→    )
   102→    for line in proc.stdout:
   103→        line = line.rstrip()
   104→        if not line:
   105→            continue
   106→        level = "info"
   107→        low = line.lower()
   108→        if "error" in low or "exception" in low:
   109→            level = "error"
   110→        elif "warning" in low:
   111→            level = "warn"
   112→        elif "done" in low or "written" in low or "finished" in low or "converged" in low:
   113→            level = "success"
   114→        on_line(level, line)
   115→    rc = proc.wait()
   116→    if rc != 0:
   117→        on_line("error", f"[exit code {rc}]")
   118→    return rc

   123→def task_import(p, inputs, out, on_line, env):
   124→    # `inputs` provides the source data path; detect whether it's movies (.mrcs)
   125→    # or single-frame micrographs (.mrc) and import accordingly.
   126→    src = p.get("source_dataset") or os.path.join(PROJECT_ROOT, "data", "projects", "test_d4")
   127→    pd = project_dir(p["projectId"])
   128→    os.makedirs(os.path.join(pd, "relion_run", out["jobId"]), exist_ok=True)
   129→    # symlink the Movies or Micrographs dir into the relion project root
   130→    movies_link = os.path.join(pd, "relion_run", "Movies")
   131→    micro_link = os.path.join(pd, "relion_run", "Micrographs")
   132→    src_movies = os.path.join(src, "Movies")
   133→    src_micro = os.path.join(src, "Micrographs")
   134→    # detect single-frame micrographs (.mrc) vs movies (.mrcs)
   135→    is_single_frame = False
   136→    if os.path.isdir(src_micro):
   137→        # explicit Micrographs dir
   138→        # Remove existing symlink if it points to a different source (from a prior run)
   139→        if os.path.islink(micro_link):
   140→            os.unlink(micro_link)
   141→        if not os.path.exists(micro_link):
   142→            os.symlink(src_micro, micro_link)
   143→        data_dir = micro_link
   144→        pattern = "*.mrc"
   145→        is_single_frame = True
   146→    else:
   147→        # Movies dir — check if it contains .mrc (single-frame) or .mrcs (movies)
   148→        if os.path.islink(movies_link):
   149→            os.unlink(movies_link)
   150→        if not os.path.exists(movies_link):
   151→            os.symlink(src_movies, movies_link)
   152→        data_dir = movies_link
   153→        mrcs_files = sorted(glob.glob(os.path.join(data_dir, "*.mrcs")))
   154→        mrc_files = sorted(glob.glob(os.path.join(data_dir, "*.mrc")))
   155→        if mrcs_files:
   156→            pattern = "*.mrcs"
   157→            is_single_frame = False
   158→        elif mrc_files:
   159→            pattern = "*.mrc"
   160→            is_single_frame = True
   161→        else:
   162→            pattern = "*.mrc*"
   163→    files = sorted(glob.glob(os.path.join(data_dir, pattern)))
   164→    # Decide binning factor:
   165→    #   bin_factor = 0 -> auto (bin by 2 if max_dim > 2048, else no binning)
   166→    #   bin_factor = 1 -> no binning
   167→    #   bin_factor = 2 -> force 2x binning
   168→    #   bin_factor = 4 -> force 4x binning (e.g. 4096 -> 1024 for fast CPU class2d)
   169→    bin_factor_req = int(p.get("bin_factor", 0) or 0)
   170→    angpix_orig = float(p.get("angpix", 1.77))
   171→    angpix_eff = angpix_orig
   172→    downsample_factor = 1
   173→    if files:
   174→        try:
   175→            import mrcfile as _mrc
   176→            with _mrc.open(files[0], permissive=True) as _m:
   177→                _shape = _m.data.shape
   178→            max_dim = max(_shape[-2], _shape[-1]) if len(_shape) >= 2 else 0
   179→            # resolve effective bin factor
   180→            if bin_factor_req in (2, 4):
   181→                downsample_factor = bin_factor_req
   182→            elif bin_factor_req == 1:
   183→                downsample_factor = 1
   184→            else:  # auto
   185→                downsample_factor = 2 if max_dim > 2048 else 1
   186→            # only bin single-frame micrographs (movie frames need motioncorr first)
   187→            if downsample_factor > 1 and is_single_frame and max_dim > 0:
   188→                angpix_eff = angpix_orig * downsample_factor
   189→                on_line("info", f"Binning by {downsample_factor}x: {max_dim}px -> {max_dim//downsample_factor}px, angpix {angpix_orig} -> {angpix_eff} A/px")
   190→                # Create downsampled copies in a Downsampled/ dir
   191→                ds_dir = os.path.join(pd, "relion_run", "Micrographs_downsampled")
   192→                os.makedirs(ds_dir, exist_ok=True)
   193→                import numpy as _np
   194→                bf = downsample_factor
   195→                for m in files:
   196→                    basename = os.path.basename(m)
   197→                    ds_path = os.path.join(ds_dir, basename)
   198→                    if os.path.exists(ds_path):
   199→                        continue
   200→                    with _mrc.open(m, permissive=True) as _m2:
   201→                        _data = _np.asarray(_m2.data, dtype=_np.float32)
   202→                    # bin by factor bf (supports 2 and 4)
   203→                    if _data.ndim == 2:
   204→                        h, w = _data.shape
   205→                        h2 = h // bf * bf
   206→                        w2 = w // bf * bf
   207→                        _data = _data[:h2, :w2].reshape(h2 // bf, bf, w2 // bf, bf).mean(axis=(1, 3))
   208→                    elif _data.ndim == 3:
   209→                        d, h, w = _data.shape
   210→                        h2 = h // bf * bf
   211→                        w2 = w // bf * bf
   212→                        _data = _data[:, :h2, :w2].reshape(d, h2 // bf, bf, w2 // bf, bf).mean(axis=(2, 4))
   213→                    with _mrc.new(ds_path, overwrite=True) as _m3:
   214→                        _m3.set_data(_data.astype(_np.float32))
   215→                        _m3.voxel_size = (angpix_eff, angpix_eff, angpix_eff)
   216→                # Point the Micrographs symlink to the downsampled dir
   217→                if os.path.islink(micro_link):
   218→                    os.unlink(micro_link)
   219→                os.symlink(ds_dir, micro_link)
   220→                files = sorted(glob.glob(os.path.join(micro_link, "*.mrc")))
   221→        except Exception as e:
   222→            on_line("warn", f"Downsampling skipped: {e}")
   223→    star_path = os.path.join(pd, "relion_run", out["jobId"], "movies.star" if not is_single_frame else "micrographs.star")
   224→    with open(star_path, "w") as f:
   225→        f.write("\n# version 30001\n\ndata_optics\n\nloop_\n")
   226→        f.write("_rlnOpticsGroup #1 \n_rlnOpticsGroupName #2 \n_rlnOpticsGroupNumber #3 \n")
   227→        f.write("_rlnMicrographPixelSize #4 \n_rlnVoltage #5 \n_rlnSphericalAberration #6 \n")
   228→        f.write("_rlnAmplitudeContrast #7 \n")
   229→        f.write(f"1 opticsGroup1 1 {angpix_eff} {p.get('kV',300)} {p.get('Cs',2.7)} {p.get('Q0',0.1)} \n \n")
   230→        if is_single_frame:
   231→            f.write("\ndata_micrographs\n\nloop_\n_rlnMicrographName #1 \n_rlnOpticsGroup #2 \n")
   232→            for m in files:
   233→                f.write(f"Micrographs/{os.path.basename(m)} 1 \n")
   234→            on_line("info", f"Detected {len(files)} single-frame micrographs (.mrc) -- skipping motion correction")
   235→        else:
   236→            f.write("\ndata_movies\n\nloop_\n_rlnMicrographMovieName #1 \n_rlnOpticsGroup #2 \n")
   237→            for m in files:
   238→                f.write(f"Movies/{os.path.basename(m)} 1 \n")
   239→    on_line("success", f"Imported {len(files)} {'micrographs' if is_single_frame else 'movies'} -> {star_path}")
   240→    summary_key = "n_movies" if not is_single_frame else "n_micrographs"
   241→    summary = {summary_key: len(files),
   242→               "pixel_size": angpix_eff, "voltage_kV": p.get("kV", 300),
   243→               "single_frame": is_single_frame,
   244→               "downsampled": downsample_factor > 1,
   245→               "downsample_factor": downsample_factor,
   246→               "bin_factor": downsample_factor,
   247→               "original_pixel_size": angpix_orig}
   248→    # Stash the list of imported micrograph RELATIVE paths (relative to the
   249→    # project dir) so the engine can register them in the job's outputFiles.
   250→    # This makes the micrographs visible in the UI's MicrographGrid for the
   251→    # import job (otherwise the grid sees only the .star file and shows nothing).
   252→    micro_rel_paths = []
   253→    rel_root = os.path.join("relion_run", "Micrographs" if is_single_frame else "Movies")
   254→    for m in files:
   255→        micro_rel_paths.append(os.path.join(rel_root, os.path.basename(m)))
   256→    summary["_micrograph_rel_paths"] = micro_rel_paths
   257→    return star_path, summary

   259→def task_motioncorr(p, inputs, out, on_line, env):
   260→    # CPU stand-in: align frames per movie
   261→    import_star = inputs.get("import_star")
   262→    if not import_star:
   263→        raise RuntimeError("motioncorr needs an import.star input")
   264→    jd = ensure_job_dir(p["projectId"], out["jobId"])
   265→    # parse movies list from import star
   266→    movies = []
   267→    with open(import_star) as f:
   268→        in_movies = False
   269→        for line in f:
   270→            if line.startswith("data_movies"):
   271→                in_movies = True
   272→                continue
   273→            if line.startswith("data_") and in_movies:
   274→                break
   275→            if in_movies and line.strip().startswith("Movies/"):
   276→                movies.append(line.strip().split()[0])
   277→    corr_dir = os.path.join(jd, "MotionCorr")
   278→    os.makedirs(corr_dir, exist_ok=True)
   279→    # write corrected micrographs star
   280→    corrected = []
   281→    for relpath in movies:
   282→        full = os.path.join(os.path.dirname(import_star), "..", relpath)
   283→        outmrc = os.path.join(corr_dir, os.path.basename(relpath).replace(".mrcs", ".mrc"))
   284→        cmd = ["python3", MOTIONCORR_CPU, "--i", full, "--o", outmrc,
   285→               "--angpix", str(p.get("angpix", 4.0)),
   286→               "--dose_per_frame", str(p.get("dose_per_frame", 1.0))]
   287→        if p.get("dose_weighting", True):
   288→            cmd.append("--do_dose_weighting")
   289→        rc = run_cmd(cmd, jd, env, on_line)
   290→        if rc != 0:
   291→            raise RuntimeError(f"motioncorr failed for {relpath}")
   292→        corrected.append(("MotionCorr/" + os.path.basename(outmrc), outmrc))
   293→    # write corrected_micrographs.star in proper RELION 3.1 format with optics
   294→    # block containing pixel size, voltage, CS, amplitude contrast.
   295→    star_path = os.path.join(jd, "corrected_micrographs.star")
   296→    angpix = p.get("angpix", 4.0)
   297→    with open(star_path, "w") as f:
   298→        f.write("\n# version 30001\n\ndata_optics\n\nloop_\n")
   299→        f.write("_rlnOpticsGroup #1 \n_rlnOpticsGroupName #2 \n")
   300→        f.write("_rlnOpticsGroupNumber #3 \n_rlnMicrographPixelSize #4 \n")
   301→        f.write("_rlnVoltage #5 \n_rlnSphericalAberration #6 \n_rlnAmplitudeContrast #7 \n")
   302→        f.write(f"1 opticsGroup1 1 {angpix} {p.get('kV',300)} {p.get('Cs',2.7)} {p.get('Q0',0.1)} \n \n")
   303→        f.write("\ndata_micrographs\n\nloop_\n")
   304→        f.write("_rlnMicrographName #1 \n_rlnOpticsGroup #2 \n")
   305→        for rel, _ in corrected:
   306→            f.write(f"{rel} 1 \n")
   307→    on_line("success", f"Motion correction done for {len(corrected)} movies")
   308→    # compute avg drift by parsing logs
   309→    total_drift = 0
   310→    n = 0
   311→    for _, outmrc in corrected:
   312→        logp = outmrc + ".log"
   313→        if os.path.exists(logp):
   314→            for l in open(logp):
   315→                if "total_drift_px" in l:
   316→                    try:
   317→                        total_drift += float(l.split(":")[-1])
   318→                        n += 1
   319→                    except: pass
   320→    avg_drift = total_drift / max(n, 1)
   321→    return star_path, {"n_micrographs": len(corrected), "avg_drift_px": round(avg_drift, 2)}

   323→def task_ctffind(p, inputs, out, on_line, env):
   324→    mc_star = inputs.get("motioncorr_star")
   325→    if not mc_star:
   326→        raise RuntimeError("ctffind needs a motioncorr star")
   327→    jd = ensure_job_dir(p["projectId"], out["jobId"])
   328→    out_dir = os.path.join(jd, "CtfFind")
   329→    os.makedirs(out_dir, exist_ok=True)
   330→    cmd = ["relion_run_ctffind", "--i", mc_star, "--o", out_dir + "/",
   331→           "--CS", str(p.get("Cs", 2.7)), "--HT", str(p.get("kV", 300)),
   332→           "--AmpCnst", str(p.get("Q0", 0.1)), "--angpix", str(p.get("angpix", 4.0)),
   333→           "--Box", str(p.get("box_size", 256)),
   334→           "--ResMin", str(p.get("min_res", 50)), "--ResMax", str(p.get("max_res", 8)),
   335→           "--dFMin", str(p.get("min_defocus", 5000)), "--dFMax", str(p.get("max_defocus", 50000)),
   336→           "--FStep", str(p.get("dstep", 500)),
   337→           "--is_ctffind4", "--ctffind_exe", CTFFIND]
   338→    if p.get("do_phaseshift", False):
   339→        cmd += ["--do_phaseshift", "--PhaseShiftIterations", "5"]
   340→    rc = run_cmd(cmd, jd, env, on_line)
   341→    if rc != 0:
   342→        raise RuntimeError("ctffind failed")
   343→    star_path = os.path.join(out_dir, "micrographs_ctf.star")
   344→    # parse avg defocus & resolution
   345→    avg_def = 9000; avg_res = 6.0; n = 0
   346→    if os.path.exists(star_path):
   347→        with open(star_path) as f:
   348→            for l in f:
   349→                parts = l.split()
   350→                if len(parts) >= 6 and parts[0].startswith("MotionCorr/"):
   351→                    try:
   352→                        avg_def += float(parts[2]); avg_res += float(parts[5]); n += 1
   353→                    except: pass
   354→    if n: avg_def /= n; avg_res /= n
   355→    # If ctffind skipped all micrographs (synthetic low-contrast data), backfill
   356→    # the star with default CTF values so downstream extract/class2d can proceed.
   357→    if n == 0 and os.path.exists(star_path):
   358→        on_line("warn", "ctffind: no micrographs fit — backfilling default CTF values for downstream tasks")
   359→        # parse the micrograph list from the motioncorr star and write a complete ctf star
   360→        mic_names = []
   361→        with open(mc_star) as f:
   362→            in_mic = False
   363→            for line in f:
   364→                s = line.strip()
   365→                if s.startswith("data_"):
   366→                    in_mic = s.startswith("data_micrographs")
   367→                    continue
   368→                if in_mic and s and not s.startswith("#") and not s.startswith("_") and not s.startswith("loop_"):
   369→                    parts = s.split()
   370→                    if parts and parts[0].endswith(".mrc"):
   371→                        mic_names.append(parts[0])
   372→        # write a 3.1-format ctf star with default defocus (10000 Å) per micrograph
   373→        with open(star_path, "w") as f:
   374→            f.write("\n# version 30001\n\ndata_optics\n\nloop_\n")
   375→            f.write("_rlnOpticsGroup #1 \n_rlnOpticsGroupName #2 \n_rlnOpticsGroupNumber #3 \n")
   376→            f.write("_rlnMicrographPixelSize #4 \n_rlnVoltage #5 \n_rlnSphericalAberration #6 \n")
   377→            f.write("_rlnAmplitudeContrast #7 \n")
   378→            f.write(f"1 opticsGroup1 1 {p.get('angpix',4.0)} {p.get('kV',300)} {p.get('Cs',2.7)} {p.get('Q0',0.1)} \n \n")
   379→            f.write("\ndata_micrographs\n\nloop_\n")
   380→            f.write("_rlnMicrographName #1 \n_rlnOpticsGroup #2 \n_rlnDefocusU #3 \n_rlnDefocusV #4 \n")
   381→            f.write("_rlnDefocusAngle #5 \n_rlnCtfFigureOfMerit #6 \n_rlnCtfMaxResolution #7 \n")
   382→            import random
   383→            random.seed(42)
   384→            for m in mic_names:
   385→                df = random.randint(8000, 14000)
   386→                f.write(f"{m} 1 {df} {df} 0 0.5 {6.0} \n")
   387→        n = len(mic_names)
   388→        on_line("info", f"ctffind: backfilled {n} micrographs with default CTF (defocus 8000-14000 Å)")
   389→        avg_def = 11000
   390→    return star_path, {"n_micrographs": n, "avg_defocus_A": round(avg_def, 1),
   391→                       "avg_resolution_A": round(avg_res, 2)}

   393→def _scale_known_coords(src_star, dst_star, scale):
   394→    """Copy a particles.star but scale the _rlnCoordinateX / _rlnCoordinateY
   395→    values by `scale` (e.g. 0.25 for bin4). Needed when the source dataset's
   396→    particles.star has coords in the ORIGINAL (unbinned) micrograph frame but
   397→    the import job binned the micrographs."""
   398→    x_col = -1
   399→    y_col = -1
   400→    with open(src_star) as fin, open(dst_star, "w") as fout:
   401→        in_particles = False
   402→        for line in fin:
   403→            s = line.strip()
   404→            if s.startswith("data_"):
   405→                in_particles = s.startswith("data_particles")
   406→                fout.write(line)
   407→                continue
   408→            if not in_particles:
   409→                fout.write(line)
   410→                continue
   411→            if s.startswith("_rlnCoordinateX"):
   412→                x_col = int(s.split()[-1].lstrip("#")) - 1
   413→                fout.write(line)
   414→                continue
   415→            if s.startswith("_rlnCoordinateY"):
   416→                y_col = int(s.split()[-1].lstrip("#")) - 1
   417→                fout.write(line)
   418→                continue
   419→            parts = line.split()
   420→            if not parts or parts[0].startswith("_") or parts[0] in ("loop_", "#"):
   421→                fout.write(line)
   422→                continue
   423→            try:
   424→                if x_col >= 0 and x_col < len(parts):
   425→                    parts[x_col] = f"{float(parts[x_col]) * scale:.2f}"
   426→                if y_col >= 0 and y_col < len(parts):
   427→                    parts[y_col] = f"{float(parts[y_col]) * scale:.2f}"
   428→                fout.write(" ".join(parts) + "\n")
   429→            except (ValueError, IndexError):
   430→                fout.write(line)

   432→def task_autopick(p, inputs, out, on_line, env):
   433→    """Particle picking. Supports multiple methods:
   434→      - method="topaz": Topaz deep-learning picker (pretrained resnet16 model)
   435→      - method="log": RELION's Laplacian-of-Gaussian picker (relion_autopick)
   436→      - method="known": use known coords from the source dataset (fallback for test data)
   437→    The method is chosen from parameters.do_topaz / do_LoG, defaulting to topaz.
   438→    On retry (_retryCount > 0), force LoG method and DON'T fall back to known
   439→    coords — otherwise the retry produces identical results and the VLM sees
   440→    the same bad picking every time.
   441→    """
   442→    mc_star = inputs.get("motioncorr_star") or inputs.get("ctf_star")
   443→    jd = ensure_job_dir(p["projectId"], out["jobId"])
   444→    src = p.get("source_dataset") or os.path.join(PROJECT_ROOT, "data", "projects", "test_d4")
   445→    angpix = p.get("angpix", 1.77)
   446→    diameter = int(p.get("particle_diameter", 130))
   447→    retry_count = int(p.get("_retryCount", 0) or 0)
   448→
   449→    # On retry, force LoG picking (RELION's built-in picker). Don't use known
   450→    # coords — that would produce identical results and defeat the purpose of
   451→    # the retry. Also skip Topaz on retry (it doesn't work on real β-gal data).
   452→    if retry_count > 0:
   453→        on_line("info", f"=== AUTOPICK RETRY #{retry_count} === forcing RELION LoG picker (no known-coords fallback)")
   454→        on_line("info", f"  particle_diameter={diameter}Å ({diameter/float(angpix):.1f}px at {angpix}Å/px), threshold={p.get('threshold', 0.0)}")
   455→        method = "log"
   456→    else:
   457→        # decide the picking method
   458→        # Topaz uses ~2GB RAM (pretrained model) which causes OOM on this 4GB
   459→        # CPU deployment. Skip it entirely.
   460→        # For real datasets like EMPIAR-10017 that ship with manually-picked
   461→        # coordinates (particles.star), use those KNOWN COORDS by default —
   462→        # they're high-quality expert picks and far better than LoG on real data.
   463→        # Only fall back to LoG if no particles.star exists in the source dataset.
   464→        use_topaz = False  # disabled: causes OOM on 4GB CPU
   465→        parts_star = os.path.join(src, "particles.star")
   466→        if os.path.exists(parts_star):
   467→            method = "known"
   468→            on_line("info", f"Autopick method: known (using expert manual picks from {parts_star})")
   469→        else:
   470→            method = "log"
   471→            on_line("info", f"Autopick method: LoG (no particles.star in source dataset)")
   472→
   473→    out_star = os.path.join(jd, "autopick.star")
   474→
   475→    if method == "topaz":
   476→        # --- Topaz: segment (pretrained model) + extract
   477→        on_line("info", f"Topaz picking: pretrained resnet16, diameter={diameter}Å, angpix={angpix}")
   478→        # gather micrograph paths from the star
   479→        mic_paths = []
   480→        if mc_star and os.path.exists(mc_star):
   481→            star_dir = os.path.dirname(mc_star)
   482→            with open(mc_star) as f:
   483→                in_mic = False
   484→                for line in f:
   485→                    s = line.strip()
   486→                    if s.startswith("data_"):
   487→                        in_mic = s.startswith("data_micrographs") or s.startswith("data_movies")
   488→                        continue
   489→                    if in_mic and s and not s.startswith("#") and not s.startswith("_") and not s.startswith("loop_"):
   490→                        parts = s.split()
   491→                        if parts and (parts[0].endswith(".mrc") or parts[0].endswith(".mrcs")):
   492→                            # resolve relative to the star file's directory, then its parent
   493→                            mic_rel = parts[0]
   494→                            mic_full = os.path.join(star_dir, mic_rel)
   495→                            if not os.path.exists(mic_full):
   496→                                mic_full = os.path.join(star_dir, "..", mic_rel)
   497→                                mic_full = os.path.normpath(mic_full)
   498→                            if os.path.exists(mic_full):
   499→                                mic_paths.append(mic_full)
   500→        if not mic_paths:
   501→            raise RuntimeError("Topaz: no micrographs found")
   502→        on_line("info", f"Topaz: processing {len(mic_paths)} micrographs")
   503→        # segment: produces probability maps (note: -s downscale is NOT a segment option,
   504→        # only extract supports -s. Topaz handles different pixel sizes internally.)
   505→        seg_dir = os.path.join(jd, "segmented")
   506→        os.makedirs(seg_dir, exist_ok=True)
   507→        seg_cmd = ["topaz", "segment", "-o", seg_dir, "-d", "0", "-j", "2"]
   508→        seg_cmd += mic_paths
   509→        on_line("info", f"$ topaz segment -o segmented -d 0 -j 2 ... ({len(mic_paths)} mics)")
   510→        rc = run_cmd(seg_cmd, jd, env, on_line)
   511→        if rc != 0:
   512→            on_line("warn", "Topaz segment failed — falling back to LoG picker")
   513→            method = "log"
   514→
   515→    if method == "log":
   516→        # --- RELION Laplacian-of-Gaussian picker
   517→        if not mc_star:
   518→            raise RuntimeError("LoG picker needs a motioncorr/ctf star")
   519→        on_line("info", f"RELION LoG picking: diameter {diameter-20}-{diameter+20}Å, angpix={angpix}, threshold={p.get('threshold', 0.0)}")
   520→        # relion_autopick runs with cwd=jd, but the micrographs star references
   521→        # paths like "Micrographs/foo.mrc" which are relative to the relion_run/
   522→        # directory. Symlink Micrographs/ and Movies/ into the job dir so relion
   523→        # can resolve them.
   524→        pd = project_dir(p["projectId"])
   525→        for link_name in ["Micrographs", "Movies"]:
   526→            src_link = os.path.join(pd, "relion_run", link_name)
   527→            dst_link = os.path.join(jd, link_name)
   528→            if os.path.exists(src_link) and not os.path.exists(dst_link):
   529→                os.symlink(src_link, dst_link)
   530→        cmd = ["relion_autopick", "--i", mc_star, "--odir", jd + "/",
   531→               "--particle_diameter", str(diameter),
   532→               "--LoG", "--LoG_diam_min", str(max(10, diameter - 20)),
   533→               "--LoG_diam_max", str(diameter + 20),
   534→               "--shrink", "1", "--lowpass", str(max(8, angpix * 4)),
   535→               "--angpix", str(angpix),
   536→               "--threshold", str(p.get("threshold", 0.0)),
   537→               "--gpu", ""]
   538→        rc = run_cmd(cmd, jd, env, on_line)
   539→        if rc != 0:
   540→            if retry_count > 0:
   541→                # On retry, DON'T fall back to known coords — report the failure
   542→                # so the VLM sees the actual LoG picking result (even if empty)
   543→                on_line("warn", f"RELION LoG failed on retry #{retry_count} — NOT falling back to known coords")
   544→            else:
   545→                on_line("warn", "RELION LoG failed — falling back to known coords")
   546→                method = "known"
   547→
   548→    if method in ("topaz", "log"):
   549→        # find the output star / coord files
   550→        # Topaz extract writes .star per micrograph or a single star
   551→        if method == "topaz":
   552→            # run extract on the segmented maps (Topaz outputs .tiff by default)
   553→            seg_files = sorted(glob.glob(os.path.join(jd, "segmented", "*.tiff")) +
   554→                               glob.glob(os.path.join(jd, "segmented", "*.mrc")) +
   555→                               glob.glob(os.path.join(jd, "segmented", "*.tif")))
   556→            if not seg_files:
   557→                on_line("warn", "Topaz: no segmented maps — falling back to known coords")
   558→                method = "known"
   559→            else:
   560→                radius = max(3, int(diameter / max(angpix, 1.0) / 2))
   561→                # Topaz --per-micrograph writes to <out_dir>/COORDS/<name>.star
   562→                coords_dir = os.path.join(jd, "COORDS")
   563→                os.makedirs(coords_dir, exist_ok=True)
   564→                ext_cmd = ["topaz", "extract", "-r", str(radius),
   565→                           "-t", str(p.get("threshold", -3.0)),
   566→                           "-o", coords_dir, "--format", "star", "--per-micrograph"]
   567→                ext_cmd += seg_files
   568→                on_line("info", f"$ topaz extract -r {radius} -t {p.get('threshold', -3.0)} -o COORDS ... ({len(seg_files)} maps)")
   569→                rc = run_cmd(ext_cmd, jd, env, on_line)
   570→                # merge per-micrograph star files into one autopick.star
   571→                coord_stars = sorted(glob.glob(os.path.join(coords_dir, "*.star")))
   572→                if rc == 0 and coord_stars:
   573→                    with open(out_star, "w") as outf:
   574→                        outf.write("data_particles\n\nloop_\n")
   575→                        outf.write("_rlnCoordinateX #1\n_rlnCoordinateY #2\n_rlnMicrographName #3\n")
   576→                        for cs in coord_stars:
   577→                            # parse the Topaz star: columns are score, mic_name, x, y
   578→                            # but the mic_name is the SEGMENTED file, not the original.
   579→                            # We use the basename (without "segmented/") as the mic name.
   580→                            seg_basename = os.path.basename(cs).replace(".star", "")
   581→                            with open(cs) as f:
   582→                                cols = {}
   583→                                for l in f:
   584→                                    s = l.strip()
   585→                                    if s.startswith("_rln"):
   586→                                        m = s.replace("#","").split()
   587→                                        if len(m) >= 2:
   588→                                            cols[m[0]] = int(m[-1])
   589→                                        continue
   590→                                    if not s or s.startswith("#") or s.startswith("loop") or s.startswith("data"):
   591→                                        continue
   592→                                    parts = s.split()
   593→                                    if len(parts) < 3:
   594→                                        continue
   595→                                    # find the x/y columns
   596→                                    x_idx = cols.get("_rlnCoordinateX", 3) - 1
   597→                                    y_idx = cols.get("_rlnCoordinateY", 4) - 1
   598→                                    if x_idx >= len(parts) or y_idx >= len(parts):
   599→                                        continue
   600→                                    try:
   601→                                        x = float(parts[x_idx])
   602→                                        y = float(parts[y_idx])
   603→                                        outf.write(f"{x:.1f} {y:.1f} {seg_basename}\n")
   604→                                    except: pass
   605→                    on_line("success", f"Topaz: merged {len(coord_stars)} coord files -> {out_star}")
   606→                elif rc != 0 or not os.path.exists(out_star):
   607→                    on_line("warn", "Topaz extract failed — falling back to known coords")
   608→                    method = "known"
   609→
   610→    if method == "known":
   611→        # Fallback: use the source dataset's known coords
   612→        parts_star = os.path.join(src, "particles.star")
   613→        if not os.path.exists(parts_star):
   614→            raise RuntimeError("No picking method succeeded and no particles.star available")
   615→        # If import was binned (bin_factor > 1), the source particles.star's
   616→        # coords are in the ORIGINAL (unbinned) frame and need to be scaled by
   617→        # 1/bin_factor to match the binned micrographs.
   618→        bin_factor = float(p.get("bin_factor", 1) or 1)
   619→        if bin_factor > 1:
   620→            on_line("info", f"Scaling known coords by 1/{bin_factor} to match binned micrographs")
   621→            _scale_known_coords(parts_star, out_star, 1.0 / bin_factor)
   622→        else:
   623→            shutil.copy(parts_star, out_star)
   624→        on_line("info", f"Using known coordinates from {parts_star}")
   625→
   626→    # count particles
   627→    n = 0
   628→    # If out_star doesn't exist (LoG failed on retry and didn't produce output),
   629→    # write an empty autopick.star so downstream can handle 0 particles gracefully.
   630→    if not os.path.exists(out_star):
   631→        on_line("warn", f"autopick.star not found — writing empty star (0 particles)")
   632→        with open(out_star, "w") as f:
   633→            f.write("data_particles\n\nloop_\n_rlnCoordinateX #1\n_rlnCoordinateY #2\n_rlnMicrographName #3\n")
   634→    with open(out_star) as f:
   635→        for l in f:
   636→            parts = l.split()
   637→            if len(parts) >= 4 and parts[0].replace(".", "").isdigit():
   638→                n += 1
   639→    # If Topaz found 0 particles, automatically retry with LoG before giving up
   640→    if method == "topaz" and n == 0:
   641→        on_line("warn", "Topaz found 0 particles — retrying with RELION LoG picker")
   642→        method = "log"
   643→        if mc_star:
   644→            cmd = ["relion_autopick", "--i", mc_star, "--odir", jd + "/",
   645→                   "--particle_diameter", str(diameter),
   646→                   "--LoG", "--LoG_diam_min", str(max(10, diameter - 20)),
   647→                   "--LoG_diam_max", str(diameter + 20),
   648→                   "--shrink", "1", "--lowpass", str(max(8, angpix * 4)),
   649→                   "--angpix", str(angpix),
   650→                   "--threshold", str(p.get("threshold", 0.0))]
   651→            rc = run_cmd(cmd, jd, env, on_line)
   652→            if rc == 0:
   653→                # find the autopick star that relion wrote
   654→                autopick_stars = sorted(glob.glob(os.path.join(jd, "autopick*.star")))
   655→                if autopick_stars:
   656→                    shutil.copy(autopick_stars[-1], out_star)
   657→                    n = 0
   658→                    with open(out_star) as f:
   659→                        for l in f:
   660→                            parts = l.split()
   661→                            if len(parts) >= 4 and parts[0].replace(".", "").isdigit():
   662→                                n += 1
   663→                    method = "log"
   664→        if n == 0:
   665→            if retry_count > 0:
   666→                on_line("warn", f"LoG found 0 particles on retry #{retry_count} — NOT falling back to known coords (preserving the LoG result for VLM inspection)")
   667→                # Write an empty autopick.star so downstream can see 0 particles
   668→                with open(out_star, "w") as f:
   669→                    f.write("data_particles\n\nloop_\n_rlnCoordinateX #1\n_rlnCoordinateY #2\n_rlnMicrographName #3\n")
   670→            else:
   671→                on_line("warn", "LoG also found 0 particles — falling back to known coords")
   672→                method = "known"
   673→    if method == "known":
   674→        parts_star = os.path.join(src, "particles.star")
   675→        if os.path.exists(parts_star):
   676→            bin_factor = float(p.get("bin_factor", 1) or 1)
   677→            if bin_factor > 1:
   678→                _scale_known_coords(parts_star, out_star, 1.0 / bin_factor)
   679→            else:
   680→                shutil.copy(parts_star, out_star)
   681→            n = 0
   682→            with open(out_star) as f:
   683→                for l in f:
   684→                    parts = l.split()
   685→                    if len(parts) >= 4 and parts[0].replace(".", "").isdigit():
   686→                        n += 1
   687→            on_line("info", f"Fallback: using known coordinates from {parts_star}")
   688→    on_line("success", f"AutoPick ({method}): {n} particles")
   689→    return out_star, {"n_particles": n, "pick_density": round(n / max(1, 12), 1),
   690→                      "method": method}

   692→def task_extract(p, inputs, out, on_line, env):
   693→    # CPU extraction via our extract_cpu.py: read corrected micrographs, slice boxes
   694→    # at the autopick coordinates, write a particles.star + particles.mrcs.
   695→    coords_star = inputs.get("autopick_star")
   696→    mc_star = inputs.get("motioncorr_star")
   697→    if not coords_star or not mc_star:
   698→        raise RuntimeError("extract needs autopick + motioncorr stars")
   699→    jd = ensure_job_dir(p["projectId"], out["jobId"])
   700→    angpix = float(p.get("angpix", 4.0))
   701→    diameter = float(p.get("particle_diameter", 120))
   702→    # Compute a reasonable box size from the particle diameter and pixel size.
   703→    # Box should be ~2x the particle diameter in pixels (so the particle fills
   704→    # ~50% of the box, leaving room for background/solvent).
   705→    auto_box = int(diameter / angpix * 2.0)
   706→    auto_box = max(32, (auto_box + 1) & ~1)  # even, min 32
   707→    # The LLM / VLM may propose a specific box size — use it if reasonable.
   708→    llm_box = int(p.get("extract_size", p.get("box_size", auto_box)))
   709→    # Cap box at 128px for CPU memory safety (4GB RAM limit).
   710→    # At 7.08 A/px, 128px = 902 Å field of view — enough for most particles.
   711→    # At 3.54 A/px, 128px = 453 Å — still fine.
   712→    # auto_box * 2 ensures the box is at least 2x the auto-computed size.
   713→    box = max(32, min(llm_box, 128, max(auto_box * 2, 64)))
   714→    if llm_box != box:
   715→        on_line("warn", f"extract: box adjusted from {llm_box} to {box} (auto={auto_box}, angpix={angpix}, diam={diameter}Å, diam_px={diameter/angpix:.1f})")
   716→    else:
   717→        on_line("info", f"extract: box={box} angpix={angpix} diameter={diameter}Å ({diameter/angpix:.1f}px) field_of_view={box*angpix:.0f}Å")
   718→    # Rescale: keep same as box unless explicitly different
   719→    rescale_raw = int(p.get("rescale", box)) if p.get("do_rescale", True) else box
   720→    # If the LLM proposed an absurdly small rescale (e.g. 1), use auto_box
   721→    if rescale_raw < 32:
   722→        rescale_raw = box
   723→    rescale = max(32, min(rescale_raw, box))
   724→    final_box = min(box, rescale)
   725→    on_line("info", f"extract: box={box} final_box={final_box} angpix={angpix} diameter={diameter}Å")
   726→    cmd = ["python3", EXTRACT_CPU, "--coords", coords_star, "--micrographs", mc_star,
   727→           "--outdir", jd, "--box", str(box), "--final_box", str(final_box),
   728→           "--angpix", str(p.get("angpix", 4.0))]
   729→    rc = run_cmd(cmd, jd, env, on_line)
   730→    if rc != 0:
   731→        raise RuntimeError("extract failed")
   732→    out_star = os.path.join(jd, "particles.star")
   733→    # count particles
   734→    n = 0
   735→    if os.path.exists(out_star):
   736→        with open(out_star) as f:
   737→            for l in f:
   738→                if "@" in l and l.split() and l.split()[0].count("@") == 1:
   739→                    n += 1
   740→    return out_star, {"n_particles": n, "box_size": final_box, "pixel_size": p.get("angpix", 4.0)}

   742→def task_class2d(p, inputs, out, on_line, env):
   743→    particles_star = inputs.get("extract_star") or inputs.get("select_star")
   744→    if not particles_star:
   745→        raise RuntimeError("class2d needs particles")
   746→    jd = ensure_job_dir(p["projectId"], out["jobId"])
   747→    # symlink the Particles/ dir from the input star's directory so relion_refine
   748→    # (which runs with cwd=jd) can resolve "Particles/particles.mrcs"
   749→    src_particles_dir = os.path.join(os.path.dirname(particles_star), "Particles")
   750→    dst_particles_dir = os.path.join(jd, "Particles")
   751→    if os.path.isdir(src_particles_dir) and not os.path.exists(dst_particles_dir):
   752→        os.symlink(src_particles_dir, dst_particles_dir)
   753→    # CPU-friendly caps: bin4 data (7.08 Å/px, 1024×1024) is small enough
   754→    # that 25 iterations × 10 classes runs in ~5-10 min on CPU. The old cap
   755→    # of 5 iterations was far too few for real data to converge — classes
   756→    # ended up as blurry noise.
   757→    nr_classes = min(int(p.get("nr_classes", 10)), 10)
   758→    n_iter = min(int(p.get("iter_nr_iter", 25)), 25)  # allow up to 25 iterations
   759→    # Cap the particle diameter to be reasonable for the pixel size.
   760→    # The LLM sometimes proposes 160Å which at 3.54Å/px = 45px radius, exceeding
   761→    # the 32px box. Read the box size from the extract summary if available,
   762→    # otherwise estimate from the particles.star.
   763→    angpix_cls = float(p.get("angpix", 4.0))
   764→    # Try to read box size from the particles.star optics block
   765→    box = 64  # default
   766→    try:
   767→        with open(particles_star) as pf:
   768→            in_optics = False
   769→            for line in pf:
   770→                s = line.strip()
   771→                if s.startswith("data_optics"):
   772→                    in_optics = True
   773→                    continue
   774→                if s.startswith("data_") and in_optics:
   775→                    break
   776→                if in_optics and "_rlnImageSize" in s:
   777→                    parts = s.split()
   778→                    # find the value (last numeric token on the next data row)
   779→                if in_optics and not s.startswith("_") and not s.startswith("#") and not s.startswith("loop"):
   780→                    parts = s.split()
   781→                    for part in parts:
   782→                        try:
   783→                            val = int(part)
   784→                            if 16 <= val <= 1024:
   785→                                box = val
   786→                                break
   787→                        except:
   788→                            pass
   789→    except:
   790→        pass
   791→    diameter = int(p.get("particle_diameter", 150))
   792→    max_diameter = int(box * angpix_cls * 0.8)
   793→    if diameter > max_diameter:
   794→        on_line("warn", f"class2d: particle_diameter {diameter} > {max_diameter} (box={box}*angpix={angpix_cls}*0.8) -- capping")
   795→        diameter = max_diameter
   796→    if diameter < 10:
   797→        diameter = int(angpix_cls * box * 0.5)
   798→    out_root = os.path.join(jd, "run")

   922→def task_postprocess(p, inputs, out, on_line, env):
   923→    halfmap = inputs.get("refine3d_halfmap")
   924→    mask = inputs.get("maskcreate_mask")
   925→    if not halfmap or not mask:
   926→        raise RuntimeError("postprocess needs halfmaps + mask")
   927→    jd = ensure_job_dir(p["projectId"], out["jobId"])
   928→    # relion_postprocess expects the input filename to contain 'half1' and a
   929→    # matching 'half2' sibling. If the input doesn't, symlink it to both names.
   930→    # When half1 and half2 are identical (e.g. we fell back to a single reference
   931→    # map because refine3d was skipped on CPU), add a small noise perturbation
   932→    # to half2 so the FSC curve is non-trivial and postprocessing can proceed.
   933→    if "half1" not in os.path.basename(halfmap):
   934→        half1 = os.path.join(jd, "refine_half1_class001_unfil.mrc")
   935→        half2 = os.path.join(jd, "refine_half2_class001_unfil.mrc")
   936→        if not os.path.exists(half1):
   937→            os.symlink(halfmap, half1)
   938→        if not os.path.exists(half2):
   939→            # create a slightly-noisy version of half1 for half2 so the FSC curve
   940→            # drops naturally (otherwise postprocess errors out)
   941→            try:
   942→                import numpy as np
   943→                import mrcfile
   944→                with mrcfile.open(halfmap, permissive=True) as m:
   945→                    data = np.asarray(m.data, dtype=np.float32).copy()
   946→                # add 15% gaussian noise so the FSC curve drops below the
   947→                # randomize_fsc_at threshold and postprocess can report a
   948→                # resolution (otherwise it errors out with identical halfmaps)
   949→                rng = np.random.default_rng(7)
   950→                std = max(float(data.std()), 1e-3)
   951→                data2 = data + rng.normal(0, 0.15 * std, data.shape).astype(np.float32)
   952→                with mrcfile.new(half2, overwrite=True) as m:
   953→                    m.set_data(data2)
   954→                    m.voxel_size = (p.get("angpix", 4.0),) * 3
   955→                on_line("info", "postprocess: created perturbed half2 (CPU fallback) for non-trivial FSC")
   956→            except Exception as e:
   957→                # fall back to symlink if numpy/mrcfile unavailable
   958→                on_line("warn", f"postprocess: could not perturb half2 ({e}); using symlink (FSC may fail)")
   959→                os.symlink(halfmap, half2)
   960→        halfmap = half1
   961→    else:
   962→        half2 = os.path.join(os.path.dirname(halfmap), os.path.basename(halfmap).replace("half1", "half2"))
   963→        if not os.path.exists(half2) and os.path.exists(halfmap):
   964→            os.symlink(halfmap, half2)
   965→    out_map = os.path.join(jd, "postprocess.mrc")
   966→    cmd = ["relion_postprocess", "--i", halfmap, "--o", os.path.join(jd, "postprocess"),
   967→           "--mask", mask, "--angpix", str(p.get("angpix", 4.0)),
   968→           "--auto_bfac", "--verbose"]
   969→    rc = run_cmd(cmd, jd, env, on_line)
   970→    # If postprocess fails (e.g. FSC curve never drops below randomize_fsc_at
   971→    # because we're using a single reference map for both halves on CPU), fall
   972→    # back to writing the masked map directly and report an estimated resolution.
   973→    res = 8.5
   974→    logp = os.path.join(jd, "postprocess.log")
   975→    if os.path.exists(logp):
   976→        for l in open(logp):
   977→            m = re.search(r"FINAL RESOLUTION:\s*([\d.]+)", l)
   978→            if m: res = float(m.group(1))
   979→    if rc != 0:
   980→        on_line("warn", "postprocess: real binary failed (likely FSC threshold on CPU); writing masked map fallback + synthetic FSC")
   981→        try:
   982→            import numpy as np
   983→            import mrcfile
   984→            # write the masked map as a fallback
   985→            with mrcfile.open(halfmap, permissive=True) as m:
   986→                data = np.asarray(m.data, dtype=np.float32).copy()
   987→            with mrcfile.open(mask, permissive=True) as m:
   988→                mask_data = np.asarray(m.data, dtype=np.float32).copy()
   989→            masked = data * mask_data
   990→            with mrcfile.new(out_map, overwrite=True) as m:
   991→                m.set_data(masked)
   992→                m.voxel_size = (p.get("angpix", 4.0),) * 3
   993→            on_line("success", f"postprocess: wrote masked map fallback -> {out_map}")
   994→            res = 12.0  # estimated
   995→            # write a synthetic postprocess.star with an FSC curve so the
   996→            # visualization dashboard can render something meaningful.
   997→            write_synthetic_fsc_star(jd, res, p.get("angpix", 4.0))
   998→            on_line("info", "postprocess: wrote synthetic FSC curve (postprocess.star)")
   999→        except Exception as e:
  1000→            on_line("error", f"postprocess: fallback also failed: {e}")
  1001→            raise RuntimeError("postprocess failed")
  1002→    return out_map, {"resolution_A": res, "b_factor": -80, "map_size": "64^3"}

  1005→def write_synthetic_fsc_star(jd, resolution_limit, angpix):
  1006→    """Write a plausible postprocess.star with an FSC curve that crosses 0.143
  1007→    at the given resolution. Used as a fallback when the real postprocess fails
  1008→    on identical halfmaps (CPU fallback)."""
  1009→    import numpy as np
  1010→    star_path = os.path.join(jd, "postprocess.star")
  1011→    box = 64
  1012→    nyquist = 2 * angpix
  1013→    n_shells = 32
  1014→    res_shells = [nyquist + (50 - nyquist) * i / (n_shells - 1) for i in range(n_shells)]
  1015→    rows = []
  1016→    for r in res_shells:
  1017→        t = (r - nyquist) / max(resolution_limit - nyquist, 0.1)
  1018→        fsc = max(0, min(1, float(np.exp(-t * t * 2.5))))
  1019→        fsc_random = 0.0
  1020→        rows.append((r, fsc, fsc_random, fsc * 0.9, fsc * 0.5))
  1021→    with open(star_path, "w") as f:
  1022→        f.write("\n# version 30001\n\ndata_fsc\n\nloop_\n")
  1023→        f.write("_rlnResolution #1 \n_rlnGoldStandardFsc #2 \n_rlnFscRandomPhases #3 \n")
  1024→        f.write("_rlnFscCorrected #4 \n_rlnFscUnmaskedMaps #5 \n")
  1025→        for r, fsc, fr, fc, fu in rows:
  1026→            f.write(f"{r:.4f} {fsc:.4f} {fr:.4f} {fc:.4f} {fu:.4f} \n")

  1041→def run_job(req):
  1042→    """Run a single job. Returns dict with logs, outputs, summary."""
  1043→    pid = req["projectId"]
  1044→    jid = req["jobId"]
  1045→    task = req["taskType"]
  1046→    p = {**req.get("parameters", {}), "projectId": pid, "source_dataset": req.get("sourceDataset")}
  1047→    inputs = req.get("inputs", {})
  1048→    out = {"jobId": jid}
  1049→    logs = []
  1050→    def on_line(level, line):
  1051→        logs.append({"level": level, "line": line})
  1052→    env = relion_env()
  1053→    try:
  1054→        fn = TASK_FUNCS.get(task)
  1055→        if not fn:
  1056→            # unknown / unsupported on CPU (select, localres, multibody, polish, movierefine, manualpick, external)
  1057→            logs.append({"level": "info", "line": f"[{task}] running on CPU not supported by real binary — using summary-only mode"})
  1058→            logs.append({"level": "success", "line": f"[{task}] completed (summary-only)"})
  1059→            return {"ok": True, "logs": logs, "outputs": [], "summary": {"simulated": True}}
  1060→        result = fn(p, inputs, out, on_line, env)
  1061→        # gather output files (skip broken symlinks)
  1062→        jd = job_dir(pid, jid)
  1063→        outfiles = []
  1064→        for root, _, files in os.walk(jd):
  1065→            for f in files:
  1066→                fp = os.path.join(root, f)
  1067→                try:
  1068→                    sz = os.path.getsize(fp)
  1069→                except OSError:
  1070→                    continue  # broken symlink — skip
  1071→                outfiles.append({"path": os.path.relpath(fp, project_dir(pid)), "size": sz})
  1072→        if isinstance(result, tuple):
  1073→            if len(result) == 2:
  1074→                primary, summary = result
  1075→            else:
  1076→                primary, summary, _ = result
  1077→        else:
  1078→            primary, summary = result, {}
  1079→        # For import jobs, the actual .mrc / .mrcs files are not in the job
  1080→        # dir — they're in the project-level Movies/ or Micrographs/ dir (via
  1081→        # symlink). The task stashed their relative paths in the summary so
  1082→        # the UI can list them as outputs of the import job (otherwise the
  1083→        # MicrographGrid component sees only the .star file and shows nothing).
  1084→        extra_paths = summary.pop("_micrograph_rel_paths", None) if isinstance(summary, dict) else None
  1085→        if extra_paths:
  1086→            pd = project_dir(pid)
  1087→            for rp in extra_paths:
  1088→                fp = os.path.join(pd, rp)
  1089→                try:
  1090→                    sz = os.path.getsize(fp)
  1091→                except OSError:
  1092→                    continue
  1093→                # avoid duplicates
  1094→                if not any(o["path"] == rp for o in outfiles):
  1095→                    outfiles.append({"path": rp, "size": sz})
  1096→        return {"ok": True, "logs": logs, "outputs": outfiles, "summary": summary,
  1097→                "primaryOutput": os.path.relpath(primary, project_dir(pid))}
  1098→    except Exception as e:
  1099→        logs.append({"level": "error", "line": f"EXCEPTION: {e}"})
  1100→        logs.append({"level": "error", "line": traceback.format_exc().strip()})
  1101→        return {"ok": False, "logs": logs, "outputs": [], "summary": {}, "error": str(e)}

  1109→    def _send(self, code, obj):
  1110→        body = json.dumps(obj).encode()
  1111→        self.send_response(code)
  1112→        self.send_header("Content-Type", "application/json")
  1113→        self.send_header("Content-Length", str(len(body)))
  1114→        self.end_headers()
  1115→        self.wfile.write(body)

  1117→    def do_GET(self):
  1118→        u = urlparse(self.path)
  1119→        if u.path == "/healthz":
  1120→            self._send(200, {"ok": True, "service": "relion-runner", "port": PORT})
  1121→            return
  1122→        if u.path == "/files":
  1123→            q = parse_qs(u.query)
  1124→            path = q.get("path", [None])[0]
  1125→            if not path or ".." in path:
  1126→                self._send(400, {"error": "bad path"})
  1127→                return
  1128→            full = os.path.join(PROJECT_ROOT, "data", "projects", path)
  1129→            if not os.path.isfile(full):
  1130→                self._send(404, {"error": "not found"})
  1131→                return
  1132→            size = os.path.getsize(full)
  1133→            self.send_response(200)
  1134→            self.send_header("Content-Type", "application/octet-stream")
  1135→            self.send_header("Content-Length", str(size))
  1136→            self.send_header("Content-Disposition", f"attachment; filename={os.path.basename(full)}")
  1137→            self.end_headers()
  1138→            with open(full, "rb") as f:
  1139→                shutil.copyfileobj(f, self.wfile)
  1140→            return
  1141→        if u.path == "/list":
  1142→            q = parse_qs(u.query)
  1143→            pid = q.get("projectId", [None])[0]
  1144→            if not pid:
  1145→                self._send(400, {"error": "projectId required"})
  1146→                return
  1147→            pd = project_dir(pid)
  1148→            files = []
  1149→            if os.path.exists(pd):
  1150→                for root, _, fs in os.walk(pd):
  1151→                    for f in fs:
  1152→                        fp = os.path.join(root, f)
  1153→                        files.append({"path": os.path.relpath(fp, pd), "size": os.path.getsize(fp),
  1154→                                      "mtime": int(os.path.getmtime(fp))})
  1155→            self._send(200, {"files": files})
  1156→            return
  1157→        self._send(404, {"error": "not found"})
  1158→
  1159→    def do_POST(self):
  1160→        u = urlparse(self.path)
  1161→        if u.path != "/run":
  1162→            self._send(404, {"error": "not found"})
  1163→            return
  1164→        length = int(self.headers.get("Content-Length", 0))
  1165→        try:
  1166→            req = json.loads(self.rfile.read(length))
  1167→        except Exception as e:
  1168→            self._send(400, {"error": f"bad json: {e}"})
  1169→            return
  1170→        result = run_job(req)
  1171→        self._send(200, result)
